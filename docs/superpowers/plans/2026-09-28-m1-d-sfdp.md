## Background: JESD216 (SFDP)

A chip that supports SFDP answers `5A a2 a1 a0 <8 dummy clocks>` with bytes
from a small read-only address space. DWORDs are little-endian 32-bit words.

- **Header** (address 0):
  - DW0 = `0x50444653` (`"SFDP"`)
  - DW1: minor `[7:0]`, major `[15:8]`, number of parameter headers minus
    one `[23:16]`
- **Parameter header** *i* (address `8 + 8i`):
  - DW0: id LSB `[7:0]`, minor `[15:8]`, major `[23:16]`, length in DWORDs
    `[31:24]`
  - DW1: table pointer `[23:0]`, id MSB `[31:24]`
  - The **Basic Flash Parameter Table (BFPT)** has id LSB `0x00`, MSB
    `0xFF`.
- **BFPT fields used** (DWORDs numbered from 1, as the standard does):

| DWORD | Bits | Meaning |
|---|---|---|
| 1 | `[1:0]` | `01`: 4 KiB erase supported |
| 1 | `[15:8]` | 4 KiB erase opcode |
| 1 | 16 / 20 / 21 / 22 | 1-1-2 / 1-2-2 / 1-4-4 / 1-1-4 fast read supported |
| 1 | `[18:17]` | address bytes: `00` 3 only, `01` 3 or 4, `10` 4 only |
| 1 | 19 | DTR clocking supported |
| 2 | 31, `[30:0]` | density: bit 31 = 0 → `[30:0]` + 1 bits; = 1 → 2^`[30:0]` bits |
| 3 | `[4:0]` `[7:5]` `[15:8]` | 1-4-4: dummy clocks, mode clocks, opcode |
| 3 | `[20:16]` `[23:21]` `[31:24]` | 1-1-4: same |
| 4 | low half / high half | 1-1-2 / 1-2-2: same layout |
| 5 | 0 / 4 | 2-2-2 / 4-4-4 supported |
| 6 | `[31:16]` | 2-2-2: dummy `[20:16]`, mode `[23:21]`, opcode `[31:24]` |
| 7 | `[31:16]` | 4-4-4: same |
| 8, 9 | per 16-bit half | erase types 1–4: size 2^`[7:0]` bytes (0 = unused), opcode `[15:8]` |
| 11 | `[7:4]` | page size 2^N bytes (JESD216A and later: BFPT ≥ 11 DWORDs) |

Fixtures come from QEMU's `hw/block/m25p80_sfdp.c` at commit
`81ce3a87737aa50716c42db8886082d12783e0a1` (2026-09-27). It has twelve
tables (Micron, Macronix, Winbond, ISSI), each a plain `static const uint8_t
sfdp_<model>[] = {0x.., ...};`. `hw/block/m25p80.c` maps chips to tables
with `INFO("<name>", 0x<jedec>, ...), .sfdp_read = m25p80_sfdp_<model>`
(or `INFO6`); some tables serve several chips.

```python
"""Write tests/fixtures/sfdp/*.json from QEMU's m25p80 SFDP tables at a pinned
commit. Run: uv run tools/fetch_sfdp_fixtures.py [--commit SHA]."""

from __future__ import annotations

import argparse
import json
import re
import urllib.request
from pathlib import Path

COMMIT = "81ce3a87737aa50716c42db8886082d12783e0a1"  # qemu master, 2026-09-27
RAW = "https://raw.githubusercontent.com/qemu/qemu/{commit}/hw/block/{name}"
BLOB = "https://github.com/qemu/qemu/blob/{commit}/hw/block/{name}"
OUT = Path(__file__).resolve().parent.parent / "tests" / "fixtures" / "sfdp"

TABLE = re.compile(r"static const uint8_t sfdp_(\w+)\[\] = \{(.*?)\};", re.S)
CHIP = re.compile(
    r'INFO6?\(\s*"([^"]+)",\s*(0x[0-9a-fA-F]+)(?:(?!INFO6?\().)*?\.sfdp_read\s*=\s*m25p80_sfdp_(\w+)',
    re.S,
)


def fetch(commit: str, name: str) -> str:
    with urllib.request.urlopen(RAW.format(commit=commit, name=name), timeout=60) as r:
        return str(r.read().decode("utf-8"))


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--commit", default=COMMIT)
    commit = ap.parse_args().commit
    tables_src = fetch(commit, "m25p80_sfdp.c")
    chips_src = fetch(commit, "m25p80.c")
    chips: dict[str, list[dict[str, str]]] = {}
    for name, jedec, model in CHIP.findall(chips_src):
        chips.setdefault(model, []).append({"name": name, "jedec_id": f"{int(jedec, 16):06x}"})
    OUT.mkdir(parents=True, exist_ok=True)
    for m in TABLE.finditer(tables_src):
        model, body = m.group(1), m.group(2)
        data = bytes(int(x, 16) for x in re.findall(r"0x([0-9a-fA-F]{2})", body))
        line = tables_src.count("\n", 0, m.start()) + 1
        doc = {
            "model": model,
            "chips": chips.get(model, []),
            "origin": {
                "project": "QEMU",
                "commit": commit,
                "file": "hw/block/m25p80_sfdp.c",
                "url": BLOB.format(commit=commit, name="m25p80_sfdp.c") + f"#L{line}",
                "license": "GPL-2.0-or-later",
            },
            "sfdp": data.hex(),
        }
        (OUT / f"qemu-{model}.json").write_text(json.dumps(doc, indent=1) + "\n")
        print(model, len(data), [c["name"] for c in doc["chips"]])
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""JESD216 Serial Flash Discoverable Parameters (SFDP): the header, the
parameter headers and the Basic Flash Parameter Table (BFPT). This is the
Python reference the C module (``usf_sfdp_read``) is tested against; both
read the image the same way, DWORD by DWORD, treating bytes past the end as
0xFF (what an idle bus returns)."""

from __future__ import annotations

from dataclasses import dataclass
from typing import TypedDict

from spiflash.enums import Feature
from spiflash.opcodes import OPERATIONS

SIGNATURE = 0x50444653
READ_MODES: tuple[str, ...] = ("1-1-2", "1-2-2", "1-1-4", "1-4-4", "2-2-2", "4-4-4")
_MAX_HEADERS = 16  # C reads at most this many parameter headers


@dataclass(frozen=True)
class Bfpt:
    """The BFPT fields uspiflash uses (see the table in the M1 plan, part D)."""

    major: int
    minor: int
    dwords: int
    address_bytes: int
    erase_4k_opcode: int
    size: int
    page_size: int
    erase_types: tuple[tuple[int, int], ...]
    reads: tuple[tuple[int, int], ...]


def _dword(image: bytes, addr: int) -> int:
    return int.from_bytes(image[addr : addr + 4].ljust(4, b"\xff"), "little")


def _read(dw: int, shift: int) -> tuple[int, int]:
    """(opcode, dummy + mode clocks) from a 16-bit fast-read descriptor."""
    half = dw >> shift & 0xFFFF
    return half >> 8, (half & 0x1F) + (half >> 5 & 0x7)


def parse(image: bytes) -> Bfpt | None:
    """The BFPT of an SFDP image, or None if there is none."""
    if _dword(image, 0) != SIGNATURE:
        return None
    count = min((_dword(image, 4) >> 16 & 0xFF) + 1, _MAX_HEADERS)
    for i in range(count):
        h0, h1 = _dword(image, 8 + 8 * i), _dword(image, 12 + 8 * i)
        if h0 & 0xFF == 0x00 and h1 >> 24 == 0xFF:
            break
    else:
        return None
    minor, major, length, ptr = h0 >> 8 & 0xFF, h0 >> 16 & 0xFF, h0 >> 24, h1 & 0xFFFFFF
    if length < 9:
        return None
    d = [0, *(_dword(image, ptr + 4 * k) for k in range(min(length, 11)))]  # d[1] is DWORD 1
    density = d[2]
    if density >> 31:
        log2_bits = density & 0x7FFFFFFF
        size = 1 << (log2_bits - 3) if 3 <= log2_bits < 35 else 0
    else:
        size = ((density & 0x7FFFFFFF) + 1) >> 3
    supported = (
        d[1] >> 16 & 1, d[1] >> 20 & 1, d[1] >> 22 & 1, d[1] >> 21 & 1,
        d[5] & 1, d[5] >> 4 & 1,
    )
    raw = (_read(d[4], 0), _read(d[4], 16), _read(d[3], 16), _read(d[3], 0),
           _read(d[6], 16), _read(d[7], 16))
    reads = tuple(r if ok else (0, 0) for r, ok in zip(raw, supported, strict=True))
    erase = tuple(
        (d[8 + k // 2] >> (16 * (k % 2)) & 0xFF, d[8 + k // 2] >> (16 * (k % 2) + 8) & 0xFF)
        for k in range(4)
    )
    return Bfpt(
        major=major,
        minor=minor,
        dwords=length,
        address_bytes=d[1] >> 17 & 3,
        erase_4k_opcode=d[1] >> 8 & 0xFF if d[1] & 3 == 1 else 0xFF,
        size=size,
        page_size=1 << (d[11] >> 4 & 0xF) if length >= 11 else 0,
        erase_types=erase,
        reads=reads,
    )


def dump(b: Bfpt | None) -> str:
    """The harness's ``S`` output: what the C module read."""
    if b is None:
        return "sfdp=none\n"
    lines = [
        f"sfdp={b.major}.{b.minor} dwords={b.dwords} addr={b.address_bytes} "
        f"erase4k=0x{b.erase_4k_opcode:02x} size={b.size} page={b.page_size}",
        "erase=" + " ".join(f"{n}:0x{op:02x}" for n, op in b.erase_types),
        "read=" + " ".join(f"{m}:0x{op:02x}/{clk}" for m, (op, clk) in
                           zip(READ_MODES, b.reads, strict=True)),
    ]
    return "\n".join(lines) + "\n"


def _op(protocol: str, opcode: int, kind: str) -> str | None:
    for name, op in OPERATIONS.items():
        if op.protocol == protocol and op.opcode == opcode and op.kind == kind and op.address_bytes == 3:
            return name
    return None


class View(TypedDict):
    """A BFPT in spiflash's vocabulary."""

    size: int
    page_size: int
    features: set[Feature]
    operations: set[str]


def spiflash_view(b: Bfpt) -> View:
    """The BFPT in spiflash's vocabulary, for comparing with the database."""
    features = {Feature.SFDP}
    ops: set[str] = set()
    sizes = {1 << n for n, _opcode in b.erase_types if n}
    for size, feature in ((4096, Feature.ERASE_4K), (32768, Feature.ERASE_32K),
                          (65536, Feature.ERASE_64K)):
        if size in sizes:
            features.add(feature)
    for n, opcode in b.erase_types:
        if n and (name := _op("1-1-0", opcode, "erase")):
            ops.add(name)
    for mode, (opcode, _clocks) in zip(READ_MODES, b.reads, strict=True):
        if opcode and (name := _op(mode, opcode, "read")):
            ops.add(name)
    if b.reads[0][0] or b.reads[1][0]:
        features.add(Feature.DUAL_READ)
    if b.reads[2][0] or b.reads[3][0]:
        features.add(Feature.QUAD_READ)
    if b.reads[5][0]:
        features.add(Feature.QPI)
    if b.address_bytes in (1, 2):
        features.add(Feature.FOUR_BYTE_ADDR)
    return {"size": b.size, "page_size": b.page_size, "features": features, "operations": ops}
```

```c
    uint8_t  read_clocks[6];      /* dummy + mode clocks for each */
```

```c
#if USF_HAVE_SFDP
/* One little-endian DWORD of SFDP space. */
static uint32_t usf__sfdp_dw(usf_bus *bus, uint32_t addr)
{
    uint8_t tx[5], rx[4];
    tx[0] = 0x5A;
    tx[1] = (uint8_t)(addr >> 16);
    tx[2] = (uint8_t)(addr >> 8);
    tx[3] = (uint8_t)addr;
    tx[4] = 0;                               /* 8 dummy clocks */
    USF_XFER(bus, tx, 5, rx, 4);
    return (uint32_t)rx[0] | (uint32_t)rx[1] << 8 | (uint32_t)rx[2] << 16 | (uint32_t)rx[3] << 24;
}

/* A 16-bit fast-read descriptor: opcode in the high byte, mode clocks in
 * bits 7:5, dummy clocks in bits 4:0. */
static void usf__sfdp_read_mode(usf_sfdp *out, uint8_t i, uint16_t half, uint8_t ok)
{
    out->read_opcode[i] = ok ? (uint8_t)(half >> 8) : 0;
    out->read_clocks[i] = ok ? (uint8_t)((half & 0x1F) + ((half >> 5) & 7)) : 0;
}

uint8_t usf_sfdp_read(usf_bus *bus, usf_sfdp *out)
{
    uint32_t h0 = 0, h1 = 0, d1, d2, d3, d4, d5, dw;
    uint32_t ptr;
    uint8_t i, count, len, k;

    if (usf__sfdp_dw(bus, 0) != 0x50444653UL)
        return 0;
    count = (uint8_t)(usf__sfdp_dw(bus, 4) >> 16);
    count = count >= 15 ? 16 : (uint8_t)(count + 1);   /* at most 16 headers */
    for (i = 0; i < count; i++) {
        h0 = usf__sfdp_dw(bus, 8 + 8 * (uint32_t)i);
        h1 = usf__sfdp_dw(bus, 12 + 8 * (uint32_t)i);
        if ((h0 & 0xFF) == 0x00 && (h1 >> 24) == 0xFF)
            break;
    }
    if (i == count)
        return 0;
    len = (uint8_t)(h0 >> 24);
    if (len < 9)
        return 0;
    out->minor = (uint8_t)(h0 >> 8);
    out->major = (uint8_t)(h0 >> 16);
    out->dwords = len;
    ptr = h1 & 0xFFFFFFUL;

    d1 = usf__sfdp_dw(bus, ptr);
    d2 = usf__sfdp_dw(bus, ptr + 4);
    d3 = usf__sfdp_dw(bus, ptr + 8);
    d4 = usf__sfdp_dw(bus, ptr + 12);
    d5 = usf__sfdp_dw(bus, ptr + 16);

    out->address_bytes = (uint8_t)((d1 >> 17) & 3);
    out->erase_4k_opcode = (d1 & 3) == 1 ? (uint8_t)(d1 >> 8) : 0xFF;
    if (d2 >> 31) {
        uint32_t n = d2 & 0x7FFFFFFFUL;          /* 2^n bits */
        out->size = n >= 3 && n < 35 ? (uint32_t)1 << (n - 3) : 0;
    } else {
        out->size = ((d2 & 0x7FFFFFFFUL) + 1) >> 3;
    }
    usf__sfdp_read_mode(out, 0, (uint16_t)d4, (uint8_t)((d1 >> 16) & 1));          /* 1-1-2 */
    usf__sfdp_read_mode(out, 1, (uint16_t)(d4 >> 16), (uint8_t)((d1 >> 20) & 1));  /* 1-2-2 */
    usf__sfdp_read_mode(out, 2, (uint16_t)(d3 >> 16), (uint8_t)((d1 >> 22) & 1));  /* 1-1-4 */
    usf__sfdp_read_mode(out, 3, (uint16_t)d3, (uint8_t)((d1 >> 21) & 1));          /* 1-4-4 */
    usf__sfdp_read_mode(out, 4, (uint16_t)(usf__sfdp_dw(bus, ptr + 20) >> 16), (uint8_t)(d5 & 1));
    usf__sfdp_read_mode(out, 5, (uint16_t)(usf__sfdp_dw(bus, ptr + 24) >> 16), (uint8_t)((d5 >> 4) & 1));
    for (k = 0; k < 4; k++) {
        dw = usf__sfdp_dw(bus, ptr + 28 + 4 * (uint32_t)(k >> 1));
        if (k & 1)
            dw >>= 16;
        out->erase_log2[k] = (uint8_t)dw;
        out->erase_opcode[k] = (uint8_t)(dw >> 8);
    }
    out->page_size = len >= 11
        ? (uint16_t)(1u << ((usf__sfdp_dw(bus, ptr + 40) >> 4) & 0xF)) : 0;
    return 1;
}
#endif
```

Where the C and the Python must agree exactly:
- **Header count:** Python's `min(n + 1, 16)` equals C's
  `n >= 15 ? 16 : n + 1`, which also cannot overflow `uint8_t` at n = 255.
- **Page size:** the exponent is a 4-bit field, so `1u << N` is at most
  32768 and fits even SDCC's 16-bit `unsigned`.
- **Size:** both give 0 for densities of 4 GiB and up (2^n bits with
  n ≥ 35).
- **Stack:** `usf_sfdp` is 30 bytes and belongs to the caller; the reader
  itself keeps five DWORDs.

```python
def _check(paths: Sequence[str]) -> int:
    """Regenerate each file from its embedded config and compare."""
    status = 0
    for name in paths:
        text = Path(name).read_text(encoding="ascii")
        m = CONFIG_RE.search(text)
        if m is None:
            print(f"{name}: no uspiflash-config line: not generated by uspiflash?")
            status = max(status, 2)
            continue
        config = Config.from_json(json.loads(m.group(1)))
        fresh = emit.render(config)
        if fresh == text:
            print(f"{name}: up to date")
            continue
        made = re.search(r"Generated by uspiflash (\S+) from spiflash (\S+)\.", text)
        print(f"{name}: out of date")
        if made:
            print(f"  generated by uspiflash {made.group(1)}, spiflash {made.group(2)}")
        print(f"  installed    uspiflash {__version__}, spiflash {spiflash.__version__}")
        diff = difflib.unified_diff(
            text.splitlines(), fresh.splitlines(), f"{name} (committed)", f"{name} (fresh)",
            lineterm="",
        )
        for line in itertools.islice(diff, 40):
            print("  " + line)
        status = max(status, 1)
    return status
```
