`examples/linux/Makefile`. The new lines follow the existing
`LDFLAGS ?= -static` block. `CC ?= cc` stays; make always defines `CC`,

```make
# The C library, always statically linked: glibc ($(CC)), musl (musl-gcc),
# dietlibc (diet $(CC)) or uclibc-ng (a Bootlin uClibc-ng toolchain's gcc:
# make LIBC=uclibc-ng UCLIBC_CC=/path/to/x86_64-linux-gcc).
LIBC ?= glibc
ifeq ($(LIBC),musl)
override CC := musl-gcc
# musl ships without the kernel's headers: use the system's.
LIBC_CFLAGS := -isystem build/kernel-headers
LIBC_DEPS := build/kernel-headers
else ifeq ($(LIBC),dietlibc)
override CC := diet $(CC)
else ifeq ($(LIBC),uclibc-ng)
ifeq ($(UCLIBC_CC),)
$(error LIBC=uclibc-ng needs UCLIBC_CC=/path/to/x86_64-linux-gcc)
endif
override CC := $(UCLIBC_CC)
else ifneq ($(LIBC),glibc)
$(error LIBC must be glibc, musl, dietlibc or uclibc-ng)
endif
```

build/uspiflash-linux: uspiflash-linux.c build/uspiflash.h $(LIBC_DEPS)

/*
 * The smallest useful program: look one chip id up and print what the
 * library knows, through the C library's putchar. tests/test_libc.py builds
 * it against each embedded C library in src/uspiflash/toolchains.toml
 * (newlib, newlib-nano, picolibc, avr-libc, SDCC's). The library needs
 * nothing from any of them; this program needs putchar.
 *
 * Generate the header first, e.g.:
 *   uv run uspiflash generate --level describe --type nor -o uspiflash.h
 */
#include <stdint.h>
#include <stdio.h>

#define USF_IMPLEMENTATION
#include "uspiflash.h"

#if defined(__SDCC)
/* SDCC's library leaves putchar to the program: here, a UART would send
 * the character. */
int putchar(int c)
{
    return c;
}
#endif

/* USF_REENTRANT: SDCC's 8051 port calls callbacks through a pointer only
 * when they are reentrant (the generated file's comment explains). */
static void sink(void *ctx, char ch) USF_REENTRANT
{
    (void)ctx;
    putchar(ch);
}

int main(void)
{
    static const uint8_t id[3] = {0xef, 0x40, 0x18};    /* a Winbond W25Q128 */
    usf_chip chips[USF_LOOKUP_MAX];
    uint8_t n = usf_lookup(USF_FAMILY_JEDEC, id, 3, chips);
    usf_print(chips, n, 0, sink, NULL);
    return n == 0;
}

```python
#: UBSan in trap mode: needs no runtime library, so it works with musl,
#: whose toolchains ship no sanitizer runtimes (the Alpine job).
TRAP = ["-O1", "-g", "-fsanitize=undefined", "-fsanitize-trap=undefined"]


@functools.cache
def sanitize_flags(cc: str) -> list[str]:
    """:data:`SANITIZE` when ``cc`` links a program with it, else
    :data:`TRAP`."""
    with tempfile.TemporaryDirectory() as d:
        src = Path(d) / "probe.c"
        src.write_text("int main(void) { return 0; }\n")
        res = subprocess.run(
            [cc, *SANITIZE, str(src), "-o", str(Path(d) / "probe")],
            capture_output=True,
            text=True,
            check=False,
        )
    return SANITIZE if res.returncode == 0 else TRAP
```

(`TMPDIR` points into pytest's base temporary directory, via
`conftest.py`, so this writes nothing to `/tmp`.) In `Harness.build`,
`extra = [*sanitize_flags(cc), *defines, …]` replaces `[*SANITIZE, …]`.

def _cpu() -> str:
    cpuinfo = Path("/proc/cpuinfo")
    if cpuinfo.is_file():
        for line in cpuinfo.read_text(encoding="utf-8", errors="replace").splitlines():
            if line.startswith("model name"):
                return line.split(":", 1)[1].strip()
    return platform.processor() or platform.machine()


def environment(
    tools: Sequence[str] = (), seeds: Mapping[str, int] | None = None
) -> dict[str, Any]:
    """What an experiment's results record beside its numbers (spec 7.4):
    the host, the spiflash version and each upstream's commit, each tool's
    version line (``tools`` are commands, e.g. ``clang-19``), and every
    random seed the experiment used."""
    return {
        "host": {
            "python": platform.python_version(),
            "python_implementation": platform.python_implementation(),
            "system": platform.system(),
            "release": platform.release(),
            "machine": platform.machine(),
            "cpu": _cpu(),
        },
        "spiflash": {
            "version": spiflash.__version__,
            "sources": {name: info.commit for name, info in sorted(spiflash.sources().items())},
        },
        "tools": {t: version(t) for t in tools},
        "seeds": dict(seeds or {}),
    }
```

```python
"""A Sphinx extension: the ``{toolchains}`` directive, the toolchain
inventory (``src/uspiflash/toolchains.toml``) as tables, so the toolchains
page cannot drift from what the size matrix measures. Usage::

    ```{toolchains}
    ```
"""

from __future__ import annotations

from typing import TYPE_CHECKING

from docutils.parsers.rst import Directive
from tables import table

from uspiflash import toolchains

if TYPE_CHECKING:
    from docutils import nodes
    from sphinx.application import Sphinx


class ToolchainsDirective(Directive):
    """``{toolchains}``: the compilers, then the C libraries."""

    def run(self) -> list[nodes.Node]:
        """Two tables."""
        inv = toolchains.load()

        def source(tarball: str | None, packages: tuple[str, ...]) -> str:
            if tarball:
                return f"`{inv.tarballs[tarball].url}` (sha256 `{inv.tarballs[tarball].sha256}`)"
            return ", ".join(f"`{p}`" for p in packages)

        compilers = [
            [
                f"`{t.name}`" + (" (vendor fork)" if t.vendor_fork else ""),
                t.environment,
                t.family,
                ", ".join(t.cpus),
                source(t.tarball, t.packages),
            ]
            for t in inv.toolchains
        ]
        libcs = [
            [f"`{c.name}`", c.kind, f"`{' '.join(c.cc)}`", source(c.tarball, c.packages)]
            for c in inv.libcs
        ]
        return [
            table(
                ["toolchains"],
                ["Toolchain", "Environment", "Family", "CPUs", "Source"],
                compilers,
            ),
            table(["libcs"], ["C library", "Kind", "Compiler", "Source"], libcs),
        ]


def setup(app: Sphinx) -> dict[str, bool]:
    """Register the directive."""
    app.add_directive("toolchains", ToolchainsDirective)
    return {"parallel_read_safe": True, "parallel_write_safe": True}
```
