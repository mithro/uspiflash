### Task 16: `uspiflash-linux`, a spidev tool that prints what `spiflash id` prints

**Files:**
- Create: `examples/linux/uspiflash-linux.c`, `examples/linux/Makefile`,
  `examples/linux/README.md`, `tests/linux/fake_spidev.c`,
  `tests/test_linux_tool.py`

**Interfaces:**
- Consumes: `usf_probe`, `usf_print`, `usf_print_json`, `usf_sfdp_read`,
  `usf_lookup` from a header generated with `--level full --with sfdp --with sfdp_summary --with sfdp_dumps --with datasheet --with datasheets --with records --with provenance` (every extra: the tool has no size budget, so its output is byte-identical to `spiflash id`, datasheet and `sfdp:` lines, records and the JSON `sfdp` list included; Part D's `sfdp` adds `usf_sfdp_read`);
  `tests/c/simchip.h`
- Produces:
  - `uspiflash-linux [-D DEVICE] [-s HZ] [--json] [--opcodes] [--sfdp]`:
    probes `DEVICE` (default `/dev/spidev0.0`) at `HZ` (default 1000000).
    Exit codes:
    - 0: found and printed;
    - 1: a chip answered but isn't in the database (prints its id);
    - 2: nothing answered, or a usage/device error.
  - `uspiflash-linux id HEX [--method FAMILY] [--json] [--opcodes]`: offline
    lookup, byte-identical to `spiflash id HEX [--method F] [--opcodes|--json]`,
    with the same exit codes as spiflash (0 found, 1 none).
  - `uspiflash-linux --version`: the tool's and the header's versions.

Each transaction is one `SPI_IOC_MESSAGE(2)` with chip select held: a TX
transfer of `txlen` bytes, then an RX transfer of `rxlen` bytes. This
matches the library's half-duplex model exactly, and lets the fake device
see where the command ends.

```c
/* LD_PRELOAD shim: opening $FAKE_SPIDEV gives a simulated SPI flash
 * ($FAKE_SPIDEV_SPEC, tests/c/simchip.h's spec) and SPI_IOC_MESSAGE(2)
 * transactions are answered by it. Test support only. */
#define _GNU_SOURCE
#include <dlfcn.h>
#include <fcntl.h>
#include <stdarg.h>
#include <stdint.h>
#include <string.h>
#include <sys/ioctl.h>
#include <linux/spi/spidev.h>

#include "simchip.h"

static struct sim sim;
static int fake_fd = -1;

static int fake_open(const char *path, int flags, mode_t mode, const char *real_name)
{
    int (*real)(const char *, int, ...) = (int (*)(const char *, int, ...))dlsym(RTLD_NEXT, real_name);
    const char *fake = getenv("FAKE_SPIDEV");
    if (fake && strcmp(path, fake) == 0) {
        const char *spec = getenv("FAKE_SPIDEV_SPEC");
        sim.log = getenv("FAKE_SPIDEV_LOG") ? stderr : NULL;
        sim_parse(&sim, spec ? spec : "");
        fake_fd = real("/dev/null", O_RDWR);
        return fake_fd;
    }
    return real(path, flags, mode);
}

int open(const char *path, int flags, ...)
{
    mode_t mode = 0;
    if (flags & O_CREAT) {
        va_list ap;
        va_start(ap, flags);
        mode = (mode_t)va_arg(ap, int);
        va_end(ap);
    }
    return fake_open(path, flags, mode, "open");
}

int open64(const char *path, int flags, ...)
{
    mode_t mode = 0;
    if (flags & O_CREAT) {
        va_list ap;
        va_start(ap, flags);
        mode = (mode_t)va_arg(ap, int);
        va_end(ap);
    }
    return fake_open(path, flags, mode, "open64");
}

int ioctl(int fd, unsigned long request, ...)
{
    int (*real)(int, unsigned long, ...) = (int (*)(int, unsigned long, ...))dlsym(RTLD_NEXT, "ioctl");
    va_list ap;
    void *arg;
    va_start(ap, request);
    arg = va_arg(ap, void *);
    va_end(ap);
    if (fd < 0 || fd != fake_fd)
        return real(fd, request, arg);
    if (request == SPI_IOC_MESSAGE(2)) {
        struct spi_ioc_transfer *t = (struct spi_ioc_transfer *)arg;
        sim_xfer(&sim, (const uint8_t *)(uintptr_t)t[0].tx_buf, (uint8_t)t[0].len,
                 (uint8_t *)(uintptr_t)t[1].rx_buf, (uint8_t)t[1].len);
        return (int)(t[0].len + t[1].len);
    }
    return 0;  /* mode, bits per word and speed settings all "succeed" */
}
```

```c
/*
 * uspiflash-linux: identify the SPI flash chip on a Linux spidev device and
 * describe it the way `spiflash id --opcodes` does, or look an id up offline.
 *
 *   uspiflash-linux [-D /dev/spidevB.C] [-s HZ] [--json] [--opcodes] [--sfdp]
 *   uspiflash-linux id HEX [--method jedec|rems|res1|res2|at25f|st95] [--json] [--opcodes]
 *
 * Built against a header made by `uspiflash generate --level full` with
 * every extra (see the Makefile). Each library transaction is one
 * SPI_IOC_MESSAGE(2): the command bytes, then the answer, chip select held.
 */
#include <fcntl.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/ioctl.h>
#include <unistd.h>
#include <linux/spi/spidev.h>

#define USF_IMPLEMENTATION
#include "uspiflash.h"

struct dev {
    int fd;
    uint32_t hz;
};

static void xfer(void *ctx, const uint8_t *tx, uint8_t txlen, uint8_t *rx, uint8_t rxlen)
{
    struct dev *d = (struct dev *)ctx;
    struct spi_ioc_transfer t[2];
    memset(t, 0, sizeof t);
    t[0].tx_buf = (uintptr_t)tx;
    t[0].len = txlen;
    t[0].speed_hz = d->hz;
    t[0].bits_per_word = 8;
    t[1].rx_buf = (uintptr_t)rx;
    t[1].len = rxlen;
    t[1].speed_hz = d->hz;
    t[1].bits_per_word = 8;
    if (ioctl(d->fd, SPI_IOC_MESSAGE(2), t) < 0) {
        perror("uspiflash-linux: SPI_IOC_MESSAGE");
        exit(2);
    }
}

static void put(void *ctx, char ch)
{
    (void)ctx;
    putchar(ch);
}

static int family_code(const char *name)
{
    static const char *const names[] = {"jedec", "rems", "res1", "res2", "at25f", "st95"};
    int i;
    for (i = 0; i < 6; i++)
        if (strcmp(name, names[i]) == 0)
            return i;
    return -1;
}

static void hex(const uint8_t *b, unsigned n)
{
    unsigned i;
    for (i = 0; i < n; i++)
        printf("%02x", b[i]);
}

/* The chip's own SFDP, as tests/c/harness.c's S command prints it
 * (uspiflash.oracle.sfdp_fields()'s format). */
static void print_sfdp(const usf_sfdp *f)
{
    static const char *const modes[] = {"1-1-2", "1-2-2", "1-1-4", "1-4-4", "2-2-2", "4-4-4"};
    unsigned k;
    printf("sfdp=%u.%u dwords=%u addr=%u erase4k=0x%02x dtr=%u size=%lu page=%u", f->rev_major,
           f->rev_minor, f->dwords, f->address_bytes, f->erase_4k_opcode, f->flags & USF_SFDP_DTR,
           (unsigned long)f->size, f->page_size);
    if (f->quad_enable == 0xFF)
        printf(" qe=none");
    else
        printf(" qe=%u", f->quad_enable);
    printf(" en4b=0x%02x ex4b=0x%02x\nerase", f->enter_4b, f->exit_4b);
    for (k = 0; k < 4; k++)
        if (f->erase_log2[k])
            printf(" %u:%u:0x%02x", k + 1, f->erase_log2[k], f->erase_opcode[k]);
    printf("\nread");
    for (k = 0; k < 6; k++)
        if (f->reads >> k & 1)
            printf(" %s:0x%02x:%u+%u", modes[k], f->read_opcode[k], f->read_clocks[k] >> 5,
                   f->read_clocks[k] & 31u);
    putchar('\n');
}

static int usage(void)
{
    fputs("usage: uspiflash-linux [-D DEVICE] [-s HZ] [--json] [--opcodes] [--sfdp]\n"
          "       uspiflash-linux id HEX [--method FAMILY] [--json] [--opcodes]\n", stderr);
    return 2;
}

int main(int argc, char **argv)
{
    const char *device = "/dev/spidev0.0", *method = "jedec", *idhex = NULL;
    int json = 0, opcodes = 0, want_sfdp = 0, i;
    uint32_t hz = 1000000;

    for (i = 1; i < argc; i++) {
        if (!strcmp(argv[i], "-D") && i + 1 < argc)
            device = argv[++i];
        else if (!strcmp(argv[i], "-s") && i + 1 < argc)
            hz = (uint32_t)strtoul(argv[++i], NULL, 0);
        else if (!strcmp(argv[i], "--method") && i + 1 < argc)
            method = argv[++i];
        else if (!strcmp(argv[i], "--json"))
            json = 1;
        else if (!strcmp(argv[i], "--opcodes"))
            opcodes = 1;
        else if (!strcmp(argv[i], "--sfdp"))
            want_sfdp = 1;
        else if (!strcmp(argv[i], "id") && i + 1 < argc)
            idhex = argv[++i];
        else
            return usage();
    }

    if (idhex) {                          /* offline, like `spiflash id` */
        uint8_t data[64];
        unsigned n = 0, v;
        usf_chip chips[2];
        uint8_t found;
        int fam = family_code(method);
        if (fam < 0)
            return usage();
        while (n < sizeof data && idhex[0] && idhex[1] && sscanf(idhex, "%2x", &v) == 1) {
            data[n++] = (uint8_t)v;
            idhex += 2;
        }
        found = usf_lookup((uint8_t)fam, data, (uint8_t)n, chips);
        if (json)
            usf_print_json(chips, found, put, NULL);
        else
            usf_print(chips, found, opcodes ? USF_PRINT_OPCODES : 0, put, NULL);
        return found ? 0 : 1;
    }

    {                                     /* probe a real device */
        struct dev d;
        usf_bus bus;
        usf_probe_result r;
        uint8_t mode = SPI_MODE_0, bits = 8;
        d.fd = open(device, O_RDWR);
        d.hz = hz;
        if (d.fd < 0) {
            perror(device);
            return 2;
        }
        if (ioctl(d.fd, SPI_IOC_WR_MODE, &mode) < 0 || ioctl(d.fd, SPI_IOC_WR_BITS_PER_WORD, &bits) < 0
            || ioctl(d.fd, SPI_IOC_WR_MAX_SPEED_HZ, &hz) < 0) {
            perror("uspiflash-linux: configuring the spidev device");
            return 2;
        }
        bus.xfer = xfer;
        bus.ctx = &d;
        usf_probe(&bus, &r);
        if (r.count) {
            if (json)
                usf_print_json(r.chip, r.count, put, NULL);
            else
                usf_print(r.chip, r.count, opcodes ? USF_PRINT_OPCODES : 0, put, NULL);
        } else if (r.len) {
            printf("no chip in the database answers %s ", r.family == USF_FAMILY_JEDEC ? "jedec" : "legacy");
            hex(r.id, r.len);
            putchar('\n');
        } else {
            printf("no answer from the chip (the bus reads all 0x00 or all 0xff)\n");
        }
        if (want_sfdp) {
            usf_sfdp s;
            if (usf_sfdp_read(&bus, &s))
                print_sfdp(&s);
            else
                printf("sfdp=none\n");
        }
        close(d.fd);
        return r.count ? 0 : r.len ? 1 : 2;
    }
}
```

chip has a dump.

Decisions (record them in `RELEASING.md`):
- **The .deb ships a dynamically linked build.** Debian policy wants
  `Built-Using` for static linking, and a dynamic build lets
  `${shlibs:Depends}` do its job. The static build (glibc and, in M2, musl)
  remains the Makefile's default and is uploaded as a CI artifact for
  non-Debian use.
- **Mixed packages publish correctly.** An arch-specific build job uses
  `dpkg-buildpackage -b`, which also rebuilds `python3-uspiflash`
  (`all`). `publish-apt` copies every artifact's debs into one directory
  per suite, so the identically named `_all.deb`s overwrite each other.
  Checked in `mithro/apt-repo-action` at `2b86e56`,
  `.github/workflows/publish-apt.yml`, the "Regroup artifacts" step. The
  builds are reproducible (`SOURCE_DATE_EPOCH`), so they are the same file.

```python
"""uspiflash-linux inside QEMU finds QEMU's emulated flash, and its SFDP."""

from __future__ import annotations

import os
import subprocess
import sys
from pathlib import Path

import pytest
from spiflash.db import database
from spiflash.enums import IdFamily

from uspiflash import oracle
from uspiflash.levels import Selection

ROOT = Path(__file__).resolve().parent.parent
pytestmark = pytest.mark.skipif(os.environ.get("USPIFLASH_QEMU") != "1", reason="set USPIFLASH_QEMU=1")


def test_tool_in_qemu_describes_the_emulated_flash() -> None:
    res = subprocess.run([sys.executable, str(ROOT / "tools/qemu/run.py")],
                         capture_output=True, text=True, check=False, timeout=900)
    assert res.returncode == 0, res.stdout + res.stderr
    (chip,) = database().lookup("9d7019")
    assert chip.sfdp is not None
    text, _, sfdp_part = res.stdout.partition("sfdp=")
    rdid = bytes.fromhex("9d7019") + b"\xff" * 7   # what a 10-byte RDID reads from QEMU
    sel = Selection.make("full", with_=["sfdp", "sfdp_summary", "sfdp_dumps", "datasheet", "datasheets", "records", "provenance"])
    assert text == oracle.text(database(), IdFamily.JEDEC, rdid, sel, opcodes=True)
    assert "sfdp=" + sfdp_part == oracle.sfdp_fields(chip.sfdp.data)
```

What the emulated chip returns after its three id bytes (`0xff`, `0x00`,
or repeated id bytes) is QEMU's `m25p80` behaviour. Check it in Step 1,
from QEMU's source or the first run's transcript (`FAKE_SPIDEV_LOG`-style
logging isn't available in the guest, so add an `--transcript` option to
the tool that prints each transaction to stderr). Set `rdid` in the test to
what QEMU actually returns, with a comment citing the source line.
