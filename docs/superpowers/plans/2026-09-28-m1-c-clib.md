## Rules for the C template (all tasks)

- **Language:** C99, and only what SDCC accepts.
  - No `//` comments (SDCC accepts them, but the house style is `/* */`).
  - No VLAs, no designated initialisers, no compound literals, no struct
    passing or returning by value, no `inline`, no `_Bool`.
- **Types:** `uint8_t` for everything that fits, `uint16_t` for entry
  indices and offsets, `uint32_t` only for sizes.
- **Tables:** every generated table is
  `static const USF_ROM uint8_t usf__<name>[]` (an internal symbol: the
  template reads `usf__ids`, `usf__jep106`, ...). `USF_ROM` defaults to
  nothing; SDCC users define it as `__code`.
- **No compiler helpers:** no `*`, `/` or `%` by a non-constant. Multiplying
  by a constant is fine: compilers turn it into shifts and adds.
- **No `memset` or `memcpy`:** initialise arrays and structs element by
  element and copy with loops. clang at `-O0` turns an initialiser list
  such as `best[2] = {0, 0}` into a `memset` call.
- **Guards:** every static helper sits inside the `#if` of what uses it, so
  that no selection warns about unused functions or variables.
  - Code that indexes a **counted table** (`sizes`, `pages`, `sectors`,
    `volts`, `mfrs`, `ops`/`opsets`, `ext`, `conflicts`; Task 11b adds
    `dsrows`/`dslists`) sits inside `#if USF_<NAME>_COUNT`. A filtered
    selection can leave such a table without rows, and `layout.build` then
    leaves it out. Never rely on the pad byte of an empty table: gcc's
    `-Warray-bounds` sees through it at `-O2` and `-O3`.
  - The helper conditions are spelled once, as `USF__NEED_*` macros.
- **Symbols:** public symbols are `usf_*`, internal ones `usf__*`, macros
  `USF_*`. `emit.py` renames all three for `--prefix`.
- **Comments:** every public declaration has a comment saying what it
  returns and what "unknown" looks like.
- **No libc names:** never name a parameter, local or struct field after a
  libc function (`putc`, `getc`, `index`, `abs`, ...). A libc may define
  them as function-like macros, which would expand inside the library
  whenever a user includes `<stdio.h>` first. The character sink is
  therefore always `sink`.

`tests/harness.py`:

```python
"""Test helpers for the C side: generating headers and compiling them with
the strict flags. (Task 9 adds the harness driver.)"""

from __future__ import annotations

import functools
import subprocess
from typing import TYPE_CHECKING

from uspiflash import emit
from uspiflash.levels import LEVELS

if TYPE_CHECKING:
    from collections.abc import Sequence
    from pathlib import Path

    from uspiflash.provenance import Config

CFLAGS = ["-std=c99", "-Wall", "-Wextra", "-Wpedantic", "-Wundef", "-Werror"]


@functools.cache
def render(config: Config) -> str:
    """``emit.render(config)``, once per test session."""
    return emit.render(config)


def generate(tmp: Path, config: Config) -> Path:
    """Write the header ``config`` describes into ``tmp``; return its path."""
    path = tmp / config.filename
    path.write_text(render(config), encoding="ascii")
    return path


def compile_c(cc: str, sources: Sequence[str | Path], out: Path, extra: Sequence[str] = ()) -> None:
    """Compile with the strict flags; any diagnostic fails the test."""
    cmd = [cc, *CFLAGS, *extra, *(str(s) for s in sources), "-o", str(out)]
    res = subprocess.run(cmd, capture_output=True, text=True, check=False)
    assert res.returncode == 0, f"{' '.join(cmd)}\n{res.stderr}"
    assert not res.stderr, f"{' '.join(cmd)}\n{res.stderr}"


def undefined_symbols(obj: Path) -> list[str]:
    """The symbols ``obj`` needs from elsewhere (``nm -u``, names only)."""
    res = subprocess.run(
        ["nm", "-u", "--format=just-symbols", str(obj)],
        capture_output=True,
        text=True,
        check=True,
    )
    return res.stdout.split()


def name(config: Config) -> str:
    """A test id: the level, then ``+extra``, ``-removed`` and ``:type`` filters."""
    sel = config.selection
    base = LEVELS[sel.level]
    return "".join(
        [
            sel.level,
            *(f"+{f}" for f in sorted(sel.fields - base)),
            *(f"-{f}" for f in sorted(base - sel.fields)),
            *(f":{t.value}" for t in sel.chips.types),
        ]
    )
```

The C code that indexes a counted table is compiled only when the table has
rows, so `layout.build` leaves a table without rows out, and says how many
each has. A `--type nand` selection has no voltages, operations, extended
ids or conflicts. `kindnames` and `attrnames` are read only by code
indexing `ops` and `conflicts`, so they follow them. `str` is present
exactly when the template's `USF__NEED_OFF` holds: `reads_strings` states
the same condition in Python. DESCRIPTIONS alone reads it through
`usf_op_name`/`usf_op_description` (R4).

```python
"""Small helpers that write C source text."""

from __future__ import annotations


def byte_array(name: str, data: bytes, per_line: int = 16) -> str:
    """``static const USF_ROM uint8_t usf__<name>[] = {...};``: tables are
    internal symbols (``usf__*``). Never empty: C forbids a zero-length
    initialiser, so an empty table holds one unused 0."""
    body = data or b"\0"
    lines = [
        "    " + ", ".join(f"0x{b:02x}" for b in body[i : i + per_line]) + ","
        for i in range(0, len(body), per_line)
    ]
    size = f"/* {len(data)} bytes */" if data else "/* empty: nothing reads it */"
    head = f"static const USF_ROM uint8_t usf__{name}[] = {{ {size}\n"
    return head + "\n".join(lines) + "\n};\n"


def defines(values: dict[str, int]) -> str:
    """``#define USF_<NAME> <value>`` lines, sorted by name."""
    return "".join(f"#define USF_{k} {v}\n" for k, v in sorted(values.items()))
```

```python
    prefix: str = "usf"
    filename: str = "uspiflash.h"

        return {

            "filename": self.filename,
```

```python
def constants() -> dict[str, int]:
    """The C names of spiflash's vocabularies (``USF_`` is added by
    :func:`~uspiflash.cgen.defines`), generated from the tuples the layout
    indexes, so the names can never drift from the data. Enum *member*
    names throughout (``FEATURE_FOUR_BYTE_ADDR``, ``SOURCE_UBOOT``); a
    protocol's ``-`` becomes ``_`` (``PROTO_1_1_4``, ``PROTO_8D_8D_8D``)."""
    pairs = [
        *((f"FAMILY_{f.name}", i) for i, f in enumerate(FAMILIES)),
        *((f"TYPE_{t.name}", i) for i, t in enumerate(layout.TYPES)),
        *((f"FEATURE_{f.name}", i) for i, f in enumerate(layout.FEATURES)),
        ("FEATURE_COUNT", len(layout.FEATURES)),
        *((f"SOURCE_{s.name}", i) for i, s in enumerate(layout.SOURCES)),
        *((f"KIND_{k.name}", i) for i, k in enumerate(layout.KINDS)),
        *((f"PROTO_{p.replace('-', '_')}", i) for i, p in enumerate(layout.PROTOCOLS)),
        *((f"OP_{name}", i) for i, name in enumerate(layout.ALL_OPS)),
    ]
    out = dict(pairs)
    if len(out) != len(pairs):
        msg = "two constants share a name"
        raise ValueError(msg)
    return out


def _guard(filename: str) -> str:
    """The include guard: ``uspiflash.h`` gives ``USF_USPIFLASH_H``, which the
    prefix rename then makes the user's own (never a reserved identifier)."""
    return "USF_" + re.sub(r"[^A-Za-z0-9]", "_", filename).upper()


def render(config: Config, snapshot: Snapshot | None = None) -> str:
    """The generated file for ``config``."""
    if not _IDENT.fullmatch(config.prefix):
        msg = f"prefix {config.prefix!r} is not a C identifier"
        raise ValueError(msg)
    snap = snapshot or Snapshot.build(config.selection.chips.apply(database()))
    lay = layout.build(snap, config.selection)
    template = resources.files("uspiflash").joinpath("templates/uspiflash.h.in").read_text("ascii")
    code = (
        template.replace("@@GUARD@@", _guard(config.filename))
        .replace("@@CONFIG@@", defines(lay.defines) + defines(constants()))
        .replace("@@TABLES@@", "".join(byte_array(n, d) for n, d in sorted(lay.tables.items())))
    )
    if config.prefix != "usf":
        code = re.sub(r"\busf_", config.prefix + "_", code)
        code = re.sub(r"\bUSF_", config.prefix.upper() + "_", code)
    # Last, so the rename never touches the embedded configuration.
    return code.replace("@@PROVENANCE@@", header(config))
```

```c
#if USF_EXT_COUNT
/* The entry for chip id `base` given the `xl` bytes read after its id: the
 * mask of its extended ids that agree with them selects a variant. */
static uint16_t usf__narrow(uint16_t base, const uint8_t *x, uint8_t xl)
{
    usf_chip b;
    usf__p p;
    uint16_t off, mask = 0, bit = 1;
    uint8_t k, n, l, i;

    b.entry = base;
    off = usf__u16(usf__row(&b) + USF_E_EXT);
    if (!xl || off == 0xFFFF)
        return base;
    p = usf__ext + off;
    for (k = *p++; k; k--, bit = (uint16_t)(bit << 1)) {
        l = *p++;
        n = l < xl ? l : xl;
        for (i = 0; i < n && p[i] == x[i]; i++)
            ;
        if (i == n)
            mask |= bit;
        p += l;
    }
    for (k = *p++; k; k--, p += 4)
        if (usf__u16(p) == mask)
            return usf__u16(p + 2);
    return base;
}

/* The answer slot of an id header: NOR first, then NAND; one slot when the
 * file holds one type only. */
#if USF_LOOKUP_MAX == 2
#define USF__SLOT(h) (((h) >> 3) & 1)
#else
#define USF__SLOT(h) 0
#endif

uint8_t usf_lookup(uint8_t family, const uint8_t *id, uint8_t len, usf_chip *out)
{
    usf__p p = usf__ids;
    uint16_t e = 0, best[USF_LOOKUP_MAX];
    uint8_t best_len[USF_LOOKUP_MAX];
    uint8_t h, l, i, t, n = 0;

    /* Element by element: clang -O0 turns an initialiser list into memset. */
    best[0] = 0;
    best_len[0] = 0;
#if USF_LOOKUP_MAX == 2
    best[1] = 0;
    best_len[1] = 0;
#endif
    while (len > 1 && *id == 0x7f) {
        id++;
        len--;
    }
    for (; (h = *p) != 0; p += 1 + l, e++) {
        l = h & 7;
        t = USF__SLOT(h);
        if ((h >> 4) != family || l > len || l <= best_len[t])
            continue;
        for (i = 0; i < l && p[1 + i] == id[i]; i++)
            ;
        if (i == l) {
            best_len[t] = l;
            best[t] = e;
        }
    }
    for (t = 0; t < USF_LOOKUP_MAX; t++) {
        if (!best_len[t])
            continue;
        out[n].base = best[t];
#if USF_EXT_COUNT
        out[n].entry = usf__narrow(best[t], id + best_len[t], (uint8_t)(len - best_len[t]));
#else
        out[n].entry = best[t];
#endif
        n++;
    }
    return n;
}
```

```python
"""Test helpers for the C side: generating headers, compiling them with the
strict flags, and building and driving tests/c/harness.c."""

from __future__ import annotations

import functools
import random
import subprocess
from dataclasses import dataclass
from pathlib import Path
from typing import TYPE_CHECKING

from spiflash.db import database

from uspiflash import emit, layout
from uspiflash.levels import LEVELS
from uspiflash.model import FAMILIES, Snapshot, reaching_probes

if TYPE_CHECKING:
    from collections.abc import Callable, Mapping, Sequence

    from uspiflash.provenance import Config

CFLAGS = ["-std=c99", "-Wall", "-Wextra", "-Wpedantic", "-Wundef", "-Werror"]
HERE = Path(__file__).parent
SEP = "\x1e\n"
SANITIZE = ["-O1", "-g", "-fsanitize=address,undefined", "-fno-sanitize-recover=all"]


@functools.cache
def render(config: Config) -> str:
    """``emit.render(config)``, once per test session."""
    return emit.render(config)


def generate(tmp: Path, config: Config) -> Path:
    """Write the header ``config`` describes into ``tmp``; return its path."""
    path = tmp / config.filename
    path.write_text(render(config), encoding="ascii")
    return path


def compile_c(cc: str, sources: Sequence[str | Path], out: Path, extra: Sequence[str] = ()) -> None:
    """Compile with the strict flags; any diagnostic fails the test."""
    cmd = [cc, *CFLAGS, *extra, *(str(s) for s in sources), "-o", str(out)]
    res = subprocess.run(cmd, capture_output=True, text=True, check=False)
    assert res.returncode == 0, f"{' '.join(cmd)}\n{res.stderr}"
    assert not res.stderr, f"{' '.join(cmd)}\n{res.stderr}"


def undefined_symbols(obj: Path) -> list[str]:
    """The symbols ``obj`` needs from elsewhere (``nm -u``, names only)."""
    res = subprocess.run(
        ["nm", "-u", "--format=just-symbols", str(obj)],
        capture_output=True,
        text=True,
        check=True,
    )
    return res.stdout.split()


def name(config: Config) -> str:
    """A test id: the level, then ``+extra``, ``-removed`` and ``:type`` filters."""
    sel = config.selection
    base = LEVELS[sel.level]
    return "".join(
        [
            sel.level,
            *(f"+{f}" for f in sorted(sel.fields - base)),
            *(f"-{f}" for f in sorted(base - sel.fields)),
            *(f":{t.value}" for t in sel.chips.types),
        ]
    )


@functools.cache
def snapshot(config: Config) -> Snapshot:
    """The snapshot ``config``'s file is generated from (its chips only)."""
    return Snapshot.build(config.selection.chips.apply(database()))


def queries(snap: Snapshot, seed: int = 20260928) -> list[tuple[int, bytes]]:
    """Every chip id, every extended-id probe, and random near-misses."""
    rng = random.Random(seed)
    out: list[tuple[int, bytes]] = []
    for i in range(snap.n_base):
        f = snap.entries[i].flash
        fam = FAMILIES.index(f.family)
        out.append((fam, f.id))
        out.extend((fam, f.id + p) for p in reaching_probes(snap.ext.get(i, ())))
        out.append((fam, b"\x7f" * f.bank + f.id))
    out.extend(
        (rng.randrange(len(FAMILIES)), bytes(rng.randrange(256) for _ in range(rng.randint(1, 6))))
        for _ in range(3000)
    )
    return out


def _names_header() -> str:
    def arr(name: str, items: Sequence[object]) -> str:
        values = ", ".join(f'"{i}"' for i in items)
        return f"static const char *const {name}[] = {{{values}}};\n"

    return (
        arr("FAMILY_NAMES", [f.value for f in FAMILIES])
        + arr("KIND_NAMES", [k.value for k in layout.KINDS])
        + arr("PROTO_NAMES", layout.PROTOCOLS)
        + arr("FEATURE_NAMES", [f.value for f in layout.FEATURES])
        + arr("SOURCE_NAMES", [s.value for s in layout.SOURCES])
        + arr("OP_NAMES", layout.ALL_OPS)
    )


@dataclass
class Harness:
    """A compiled harness binary for one generated header."""

    binary: Path

    @classmethod
    def build(cls, tmp: Path, config: Config, cc: str) -> Harness:
        """Generate ``config``'s header in ``tmp`` and compile the harness
        against it with ``cc``, under ASan and UBSan."""
        tmp.mkdir(parents=True, exist_ok=True)
        generate(tmp, config)
        (tmp / "harness_names.h").write_text(_names_header(), encoding="ascii")
        binary = tmp / "harness"
        extra = [*SANITIZE, f"-I{tmp}", f"-I{HERE / 'c'}", f'-DUSF_HEADER="{config.filename}"']
        compile_c(cc, [HERE / "c" / "harness.c"], binary, extra)
        return cls(binary)

    def run(self, commands: Sequence[str]) -> list[str]:
        """Run ``commands`` in one process; one output per command."""
        res = subprocess.run(
            [str(self.binary)],
            input="".join(c + "\n" for c in commands),
            capture_output=True,
            text=True,
            check=False,
            timeout=600,
        )
        assert res.returncode == 0, res.stderr
        assert not res.stderr, res.stderr
        parts = res.stdout.split(SEP)
        assert parts[-1] == ""
        assert len(parts) == len(commands) + 1
        return parts[:-1]


def check(
    tmp: Path,
    compilers: Sequence[str],
    config: Config,
    oracles: Mapping[str, Callable[[int, bytes], str]],
) -> None:
    """For each compiler, build the harness for ``config``; run each harness
    command in ``oracles`` on every query of its snapshot; and compare each
    output with the oracle's ``(family, data)`` answer."""
    qs = queries(snapshot(config))
    for cc in compilers:
        h = Harness.build(tmp / cc, config, cc)
        for cmd, oracle in oracles.items():
            got = h.run([f"{cmd} {fam} {data.hex()}" for fam, data in qs])
            for (fam, data), out in zip(qs, got, strict=True):
                want = oracle(fam, data)
                assert out == want, (
                    f"{cc} {name(config)} {cmd} {fam} {data.hex()}\n--- want\n{want}--- got\n{out}"
                )
```

```python
def accessors(flash: Flash, sel: Selection) -> str:
    """The harness's ``A`` dump for one answer."""
    has = sel.has
    out = [f"family={flash.family} type={flash.type} bank={flash.bank} id={flash.id_hex}"]
    if has(Field.SIZE):
        out.append(f"size={flash.size or 0}")
    if has(Field.PAGE_SIZE):
        out.append(f"page={flash.page_size or 0}")
    if has(Field.SECTOR_SIZE):
        out.append(f"sector={flash.sector_size or 0}")
    if has(Field.VOLTAGE):
        v = flash.voltage
        out.append(f"volt={v[0]}-{v[1]}" if v else "volt=none")
    if has(Field.FEATURES):
        out.append("features=" + " ".join(f for f in layout.FEATURES if f in flash.features))
    if has(Field.MANUFACTURER):
        out.append(f"mfr={flash.manufacturer or '?'}")
    if has(Field.NAMES):
        out.append("names=" + ",".join(flash.names))
    if has(Field.SOURCES):
        out.append("sources=" + ",".join(flash.sources))
    if has(Field.OPERATIONS):
        for o in flash.opcodes.values():
            op = o.operation
            if op.kind not in sel.op_kinds:
                continue
            dummy = "var" if op.dummy_clocks is None else str(op.dummy_clocks)
            src = ",".join(o.sources) if has(Field.SOURCES) else ""
            desc = f" desc={op.description}" if has(Field.DESCRIPTIONS) else ""
            out.append(
                f"op={op.name} 0x{op.opcode:02x} kind={op.kind} proto={op.protocol} "
                f"addr={op.address_bytes} dummy={dummy} data={_DATA[op.data]} "
                f"bytes={op.data_bytes or 0} src={src}{desc}"
            )
```

`spiflash id`'s text (`spiflash/cli.py`, `describe()` and `opcode_table()`),
which the printer must reproduce character for character:

```
<ident>  <manufacturer or ?>  <names joined by ", ">  (<type>)
    size <human>[, page <human>][, sector <human>][, <v>-<v> V]
    features: <sorted features joined by " ">          (only if any)
    sources disagree on <attr>: <v> (<srcs>); <v> (<srcs>)   (per conflict, attr order size, page_size, sector_size, voltage)
    from: <sources joined by ", ">
    datasheet: <url of the best datasheet>             (only if any; Task 11b)
    opcodes:                                           (with --opcodes)
    0x<oo>  <NAME padded to the longest name>  <description>  [<sources>]
```

- `<ident>` is the `jedec_id` (`7f` × bank, then the id in hex) for the
  JEDEC family, else `<family>:<id hex>`.
- `human(n)` is `n >> 30` + ` GiB` if `n` is a whole number of GiB; else
  MiB, else KiB (`n >= scale and n % scale == 0`); else `n` + ` B`. An
  unknown size prints `?`.
- A voltage prints `%g` of volts: `2700` → `2.7`, `1650` → `1.65`,
  `3000` → `3`.
- A conflict value prints as the int, or `(2700, 3600)` for a voltage.
- A chip with no operations prints
  `    no opcodes known (the sources describe none for this part)`.
- Answers are separated by one blank line.
- At `describe` (no SOURCES/CONFLICTS), the `from:` and
  `sources disagree` lines and the `  [..]` suffixes are absent;
  `oracle.text` removes them from spiflash's output the same way.

```python
def text(db: Database, family: IdFamily, data: bytes, sel: Selection, *, opcodes: bool) -> str:
    """What `spiflash id <data> --method <family> [--opcodes]` prints, less
    what ``sel`` leaves out."""
    blocks = []
    for f in db.lookup(data, method=family):
        lines = []
        for line in describe(f, opcodes=opcodes).split("\n"):
            if line.startswith("    from: ") and not sel.has(Field.SOURCES):
                continue
            if line.startswith("    sources disagree on ") and not sel.has(Field.CONFLICTS):
                continue
            if line.startswith("    datasheet: ") and not sel.has(Field.DATASHEET):
                continue
            kept = line
            if line.startswith("    0x") and not sel.has(Field.SOURCES):
                kept = line[: line.rindex("  [")]
            lines.append(kept)
        blocks.append("\n".join(lines))
    return "\n\n".join(blocks) + "\n"
```

```c
/* Decimal without division: subtract powers of ten, so CPUs with no divide
 * instruction pull in no libgcc helper. */
static const USF_ROM uint32_t usf__p10[10] = {
    1000000000UL, 100000000UL, 10000000UL, 1000000UL, 100000UL,
    10000UL, 1000UL, 100UL, 10UL, 1UL,

static void usf__dec(uint32_t v, usf_putc_fn sink, void *ctx)
{
    uint8_t i, started = 0;
    char d;
    for (i = 0; i < 10; i++) {
        for (d = '0'; v >= usf__p10[i]; d++)
            v -= usf__p10[i];
        if (d != '0' || started || i == 9) {

            started = 1;
        }
    }
}

static void usf__human(uint32_t n, usf_putc_fn sink, void *ctx)
{
    static const USF_ROM char units[] = "GMK";
    uint8_t u, shift = 30;
    for (u = 0; u < 3; u++, shift = (uint8_t)(shift - 10)) {
        uint32_t scale = (uint32_t)1 << shift;
        if (n >= scale && !(n & (scale - 1))) {
            usf__dec(n >> shift, sink, ctx);
            sink(ctx, ' ');
            sink(ctx, units[u]);
            usf__lit("iB", sink, ctx);
            return;
        }
    }
    usf__dec(n, sink, ctx);
    usf__lit(" B", sink, ctx);
}

/* Millivolts as Python's "%g" of volts: 2700 -> 2.7, 3000 -> 3. */
static void usf__volt(uint16_t mv, usf_putc_fn sink, void *ctx)
{
    uint8_t whole = 0;
    char d;
    while (mv >= 1000) {
        mv = (uint16_t)(mv - 1000);
        whole++;
    }
    usf__dec(whole, sink, ctx);
    if (mv)
        sink(ctx, '.');
    while (mv) {           /* hundreds, tens, units, stopping at trailing zeros */
        for (d = '0'; mv >= 100; d++)
            mv = (uint16_t)(mv - 100);
        sink(ctx, d);
        mv = (uint16_t)(mv * 10);
    }
}
```

```c
    usf_putc_fn sink;
    void *ctx;
    uint8_t depth;
    uint8_t first;
} usf__json;

static void usf__jnl(usf__json *j)
{
    uint8_t i;
    j->sink(j->ctx, '\n');
    for (i = 0; i < j->depth; i++)
        j->sink(j->ctx, ' ');
}

static void usf__jitem(usf__json *j)
{
    if (!j->first)
        j->sink(j->ctx, ',');
    usf__jnl(j);
    j->first = 0;
}

{
    j->sink(j->ctx, ch);
    j->depth++;
    j->first = 1;
}

{
    j->depth--;
    if (!j->first)
        usf__jnl(j);
    j->sink(j->ctx, ch);
    j->first = 0;
}

{

    j->sink(j->ctx, '"');
```

- `tests/c/simchip.h`: `struct sim { ... }`,
  `void sim_parse(struct sim *s, const char *spec)`,
  `unsigned sim_hex(const char *s, uint8_t *buf, unsigned max)` and
  `void sim_xfer(void *ctx, const uint8_t *tx, uint8_t txlen, uint8_t *rx, uint8_t rxlen)`.
  Each transfer is logged to a caller-supplied FILE as
  `> <tx hex> < <rx hex>`.

**The sim spec:** space-separated `CC/N=HEX` items. The chip answers `HEX`
(then `fill`) to a transaction whose first byte is `CC` and which sends
`N` bytes. Also `fill=FF`, and `sfdp=HEX` (the SFDP image `0x5A` reads
from; part D). Example: a Winbond W25Q128 is `9f/1=ef4018 fill=ff`.

**Probe algorithm** (the M1 index's decision 5, and its single-type
rules). `usf_probe` compiles each step only when some chip in the file
answers it:
1. RES, `ab 00 00 00` → keep 2 bytes `res[2]` (a local). Compiled when the
   file has a NOR chip (it also wakes one from deep power-down) or a
   RES1/RES2 one.
2. RDID, `9f` → `USF_RDID_LEN` bytes into `r->id`. Compiled when the file
   has a JEDEC-family chip, NAND included: three GD5F NAND chips answer
   read-id without a dummy byte. If the answer is not all-0x00/all-0xFF:
   - look up as JEDEC; return if found;
   - remember the id (copy it to a local `rdid[USF_RDID_LEN]` and set
     `saved`).
3. NAND RDID, `9f 00` → `USF_RDID_LEN - 1` bytes; if not blank, look up as
   JEDEC; return if found. Compiled when the file has a NAND chip.
4. Only if step 2's answer was blank (`!saved`; M1 index decision 8), each
   legacy step whose family the file has:
   - REMS, `90 00 00 00` → 2 bytes; if not blank, look up as REMS; return
     if found.
   - If `res` isn't blank: copy it into `r->id` and look up 2 bytes as
     RES2, then 1 byte as RES1; return the first found (with
     `r->id`/`len`/`family` set to what matched). The lookup reads
     `r->id`, not `res`: inlined at `-O3`, gcc cannot see that the
     extended-id narrowing stays inside a two-byte array.
   - AT25F, `15` → 2 bytes; ST95, `83 00 00` → 3 bytes. Return if found.
5. Nothing found: if `saved`, restore `rdid` into `r` as the JEDEC answer
   with `count = 0` (an unknown chip); else `len = 0` (the bus is blank).
   Return 0.

"Blank" means every byte is 0x00, or every byte is 0xFF. A NAND-only
header therefore sends `9f` and `9f 00`; a NOR-only one sends `ab`, `9f`
and (to a silent bus) the legacy commands of the families it has.

```python
ID_COMMANDS = {"ab", "9f", "90", "15", "83"}
_LEGACY = {
    IdFamily.REMS: "90/4={}",  # REMS: 3 address bytes, then the id
    IdFamily.RES2: "ab/4={}",  # RES: 3 dummy bytes, then 2 id bytes
    IdFamily.RES1: "ab/4={}00",  # ...of which RES1 is the first; 00 keeps RES2 from matching
    IdFamily.AT25F: "15/1={}",
    IdFamily.ST95: "83/3={}00",  # M95: 2 address bytes, then 3 bytes
}
PROBES = {
    "all": Config(Selection.make("full")),
    "nor": Config(Selection.make("full", chips=ChipFilter(types=(FlashType.NOR,)))),
    "nand": Config(Selection.make("full", chips=ChipFilter(types=(FlashType.NAND,)))),
}


def spec_for(f: Flash, *, nand_dummy: bool) -> str:
    """A simulated chip answering the way ``f`` does. SPI NAND chips answer
    read-id after one dummy byte, or (a few) straight away: ``nand_dummy``
    picks which."""
    ident = f.id.hex()
    if f.family is not IdFamily.JEDEC:
        reply = _LEGACY[f.family].format(ident)
    elif f.type is FlashType.NAND:
        reply = f"9f/{2 if nand_dummy else 1}={ident}"
    else:
        reply = f"9f/1={'7f' * f.bank}{ident}"
    return f"{reply} fill=ff"


@pytest.fixture(scope="module")
def harnesses(tmp_path_factory: pytest.TempPathFactory, compilers: list[str]) -> dict[str, Harness]:
    """One harness per probe config, built with the first compiler found."""
    return {
        k: Harness.build(tmp_path_factory.mktemp(k), c, compilers[0]) for k, c in PROBES.items()
    }


def sent(out: str) -> list[str]:
    """The transactions a probe sent, as hex, in order."""
    return [line.split()[1] for line in out.splitlines() if line.startswith("> ")]


@pytest.mark.parametrize("which", list(PROBES))
@pytest.mark.parametrize("nand_dummy", [True, False])
def test_every_chip_is_found(
    harnesses: dict[str, Harness], which: str, *, nand_dummy: bool
) -> None:
    snap = snapshot(PROBES[which])
    chips = [snap.entries[i].flash for i in range(snap.n_base)]
    outs = harnesses[which].run([f"P {spec_for(f, nand_dummy=nand_dummy)}" for f in chips])
    for f, out in zip(chips, outs, strict=True):
        lines = out.splitlines()
        bases = [int(line.split()[2]) for line in lines if line.startswith("chip ")]
        assert any(snap.entries[b].flash.id == f.id for b in bases), (
            f"{f.family}:{f.id_hex} not found\n{out}"
        )
        assert {tx[:2] for tx in sent(out)} <= ID_COMMANDS


@pytest.mark.parametrize(
    ("which", "steps"),
    [
        ("all", ["ab000000", "9f", "9f00", "90000000", "15", "830000"]),
        ("nor", ["ab000000", "9f", "90000000", "15", "830000"]),
        # Plain 9f too: three GD5F NAND chips answer read-id without a dummy byte.
        ("nand", ["9f", "9f00"]),
    ],
)
def test_a_blank_bus_gets_each_compiled_step_once(
    harnesses: dict[str, Harness], which: str, steps: list[str]
) -> None:
    (out,) = harnesses[which].run(["P fill=ff"])
    assert sent(out) == steps
    assert "len=0" in out
    assert "count=0" in out


def test_unknown_jedec_id_is_reported(harnesses: dict[str, Harness]) -> None:
    (out,) = harnesses["all"].run(["P 9f/1=ee7777ee7777 fill=ff"])
    status = next(line for line in out.splitlines() if line.startswith("family="))
    assert status.startswith("family=jedec ")
    assert " id=ee7777ee7777" in status
    assert status.endswith(" count=0")


def test_unknown_chip_gets_only_read_id(harnesses: dict[str, Harness]) -> None:
    # A chip that answers read-id but isn't in the database must not be sent
    # REMS, AT25F or M95 commands (spec §5.3; M1 index decision 8).
    (out,) = harnesses["all"].run(["P 9f/1=ee7777ee7777 fill=ff"])
    assert {tx[:2] for tx in sent(out)} == {"ab", "9f"}


def test_res_wakes_before_rdid(harnesses: dict[str, Harness]) -> None:
    (out,) = harnesses["all"].run(["P 9f/1=ef4018 fill=ff"])
    assert sent(out)[0] == "ab000000"
```

```c
/* A simulated SPI flash for the tests: answers id commands from a table
 * (a "spec", see tests/test_probe.py) and RDSFDP (0x5A) from an SFDP image,
 * logging every transaction. Shared by tests/c/harness.c and
 * tests/linux/fake_spidev.c. */
#ifndef SIMCHIP_H
#define SIMCHIP_H

#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct sim_reply {
    uint8_t cmd, txlen, len;
    uint8_t data[16];
};

struct sim {
    struct sim_reply replies[8];
    unsigned nreplies;
    uint8_t fill;
    uint8_t sfdp[4096];
    unsigned sfdp_len;
    FILE *log;                    /* where to log transactions, or NULL */
};

static unsigned sim_hex(const char *s, uint8_t *buf, unsigned max)
{
    unsigned n = 0, v;
    while (n < max && s[0] && s[1] && sscanf(s, "%2x", &v) == 1) {
        buf[n++] = (uint8_t)v;
        s += 2;
    }
    return n;
}

/* Parse "CC/N=HEX ... fill=FF sfdp=HEX" into s (s->log is left alone). */
static void sim_parse(struct sim *s, const char *spec)
{
    static char buf[16384];
    char *tok;
    FILE *log = s->log;
    memset(s, 0, sizeof *s);
    s->log = log;
    s->fill = 0xFF;
    strncpy(buf, spec, sizeof buf - 1);
    buf[sizeof buf - 1] = 0;
    for (tok = strtok(buf, " "); tok; tok = strtok(NULL, " ")) {
        unsigned cmd, txlen;
        char hex[64];
        if (!strncmp(tok, "fill=", 5)) {
            s->fill = (uint8_t)strtoul(tok + 5, NULL, 16);
        } else if (!strncmp(tok, "sfdp=", 5)) {
            s->sfdp_len = sim_hex(tok + 5, s->sfdp, sizeof s->sfdp);
        } else if (s->nreplies < 8 && sscanf(tok, "%2x/%u=%63s", &cmd, &txlen, hex) == 3) {
            struct sim_reply *r = &s->replies[s->nreplies++];
            r->cmd = (uint8_t)cmd;
            r->txlen = (uint8_t)txlen;
            r->len = (uint8_t)sim_hex(hex, r->data, sizeof r->data);
        }
    }
}

static void sim_log(FILE *f, const uint8_t *b, uint8_t n)
{
    uint8_t i;
    for (i = 0; i < n; i++)
        fprintf(f, "%02x", b[i]);
}

static void sim_xfer(void *ctx, const uint8_t *tx, uint8_t txlen, uint8_t *rx, uint8_t rxlen)
{
    struct sim *s = (struct sim *)ctx;
    unsigned i, k;
    for (i = 0; i < rxlen; i++)
        rx[i] = s->fill;
    if (tx[0] == 0x5A && txlen == 5 && s->sfdp_len) {
        /* RDSFDP: 3 address bytes, 8 dummy clocks (the fifth byte). */
        unsigned addr = (unsigned)tx[1] << 16 | (unsigned)tx[2] << 8 | tx[3];
        for (i = 0; i < rxlen; i++)
            rx[i] = addr + i < s->sfdp_len ? s->sfdp[addr + i] : 0xFF;
    } else {
        for (k = 0; k < s->nreplies; k++)
            if (s->replies[k].cmd == tx[0] && s->replies[k].txlen == txlen) {
                for (i = 0; i < rxlen && i < s->replies[k].len; i++)
                    rx[i] = s->replies[k].data[i];
                break;
            }
    }
    if (s->log) {
        fputs("> ", s->log);
        sim_log(s->log, tx, txlen);
        fputs(" < ", s->log);
        sim_log(s->log, rx, rxlen);
        fputc('\n', s->log);
    }
}

#endif
```

```c
#if USF_HAVE_PROBE
/* Which steps some chip in this file answers (the M1 index's single-type
 * rules): RES wakes NOR chips, so it goes out whenever there is one. */
#define USF__PROBE_RES (USF_HAVE_NOR || USF_HAVE_FAMILY_RES1 || USF_HAVE_FAMILY_RES2)
#define USF__PROBE_LEGACY (USF_HAVE_FAMILY_REMS || USF_HAVE_FAMILY_RES1 \
    || USF_HAVE_FAMILY_RES2 || USF_HAVE_FAMILY_AT25F || USF_HAVE_FAMILY_ST95)

/* Whether n bytes are all 0x00 or all 0xFF: nothing answered. */
static uint8_t usf__blank(const uint8_t *b, uint8_t n)
{
    uint8_t i, any = 0, all = 0xFF;
    for (i = 0; i < n; i++) {
        any |= b[i];
        all &= b[i];
    }
    return any == 0 || all == 0xFF;
}

/* Send `tx`, keep the `rxlen` bytes that come back in r, and look them up
 * as `family` unless the bus stayed blank. Returns r->count. */
static uint8_t usf__ask(usf_bus *bus, usf_probe_result *r, uint8_t family,
                        const uint8_t *tx, uint8_t txlen, uint8_t rxlen)
{
    USF_XFER(bus, tx, txlen, r->id, rxlen);
    r->family = family;
    r->len = rxlen;
    r->count = usf__blank(r->id, rxlen) ? 0 : usf_lookup(family, r->id, rxlen, r->chip);
    return r->count;
}

uint8_t usf_probe(usf_bus *bus, usf_probe_result *r)
{
    uint8_t tx[4], saved = 0;
#if USF__PROBE_RES
    uint8_t res[2];
#endif
#if USF_HAVE_FAMILY_JEDEC
    uint8_t rdid[USF_RDID_LEN], i;
#endif

    tx[1] = 0;                          /* address or dummy bytes */
    tx[2] = 0;
    tx[3] = 0;
#if USF__PROBE_RES
    tx[0] = 0xAB;                       /* RES: also wakes from deep power-down */
    USF_XFER(bus, tx, 4, res, 2);
#endif

    tx[0] = 0x9F;                       /* JEDEC read-id */
#if USF_HAVE_FAMILY_JEDEC
    if (usf__ask(bus, r, USF_FAMILY_JEDEC, tx, 1, USF_RDID_LEN))
        return r->count;
    if (!usf__blank(r->id, USF_RDID_LEN)) {
        saved = 1;
        for (i = 0; i < USF_RDID_LEN; i++)
            rdid[i] = r->id[i];
    }
#endif
#if USF_HAVE_NAND
    /* SPI NAND: read-id after one dummy (or address) byte. The same command,
     * so it is safe for any chip; a NAND chip answers plain 0x9F with a
     * dummy byte first, which is why the first answer can be non-blank. */
    if (usf__ask(bus, r, USF_FAMILY_JEDEC, tx, 2, USF_RDID_LEN - 1))
        return r->count;
#endif

#if USF__PROBE_LEGACY
    /* Legacy id commands go only to a chip that ignored read-id: one that
     * answered it but isn't in the database is sent nothing else. */
    if (!saved) {
#if USF_HAVE_FAMILY_REMS
        tx[0] = 0x90;                   /* REMS: 3 address bytes */
        if (usf__ask(bus, r, USF_FAMILY_REMS, tx, 4, 2))
            return r->count;
#endif
#if USF_HAVE_FAMILY_RES2 || USF_HAVE_FAMILY_RES1
        if (!usf__blank(res, 2)) {      /* the RES answer: two bytes, then one */
            /* Looked up from r->id, not res: inlined at -O3, gcc cannot see
             * that narrowing stays inside a two-byte array. */
            r->id[0] = res[0];
            r->id[1] = res[1];
#if USF_HAVE_FAMILY_RES2
            r->family = USF_FAMILY_RES2;
            r->len = 2;
            if ((r->count = usf_lookup(USF_FAMILY_RES2, r->id, 2, r->chip)) != 0)
                return r->count;
#endif
#if USF_HAVE_FAMILY_RES1
            r->family = USF_FAMILY_RES1;
            r->len = 1;
            if ((r->count = usf_lookup(USF_FAMILY_RES1, r->id, 1, r->chip)) != 0)
                return r->count;
#endif
        }
#endif
#if USF_HAVE_FAMILY_AT25F
        tx[0] = 0x15;                   /* Atmel AT25F read-id */
        if (usf__ask(bus, r, USF_FAMILY_AT25F, tx, 1, 2))
            return r->count;
#endif
#if USF_HAVE_FAMILY_ST95
        tx[0] = 0x83;                   /* ST M95 read-id: 2 address bytes */
        if (usf__ask(bus, r, USF_FAMILY_ST95, tx, 3, 3))
            return r->count;
#endif
    }
#endif

    /* Nothing known answered: report the JEDEC bytes, if there were any. */
    r->family = USF_FAMILY_JEDEC;
    r->count = 0;
    r->len = 0;
#if USF_HAVE_FAMILY_JEDEC
    if (saved) {
        r->len = USF_RDID_LEN;
        for (i = 0; i < USF_RDID_LEN; i++)
            r->id[i] = rdid[i];
    }
#endif
    return 0;
}
#endif
```
