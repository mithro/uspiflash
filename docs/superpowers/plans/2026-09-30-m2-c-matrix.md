in `.worktrees/m2c`, from `origin/main` after M2b merged.

**What this part leaves behind:**
- `src/uspiflash/toolchains.toml` lists every toolchain and C library;
  `uspiflash toolchains list` prints them.
- `uspiflash toolchains run ENV -- CMD` runs a command in an
  environment's container, set up and limited by the sandbox.
- `uspiflash measure --matrix ENV --write|--check` keeps
  `sizes/matrix/ENV.json` and `sizes/matrix/README.md` current.
- `matrix.yml` checks them on pull requests, main and nightly. It comments
  size changes on PRs, and opens an issue when the nightly check fails.
- `tools/images/` can publish each environment to ghcr.io, but nothing
  needs it.

### Task 10: The inventory and `uspiflash toolchains list`

**Files:**
- Create: `src/uspiflash/toolchains.toml`, `src/uspiflash/toolchains.py`,
  `tests/test_toolchains.py`
- Modify: `src/uspiflash/cli.py` (`toolchains list`)

**Interfaces:**
- Consumes: `measure.CPUS`, `measure.target`, `measure.family_of`,
  `ledger.IMAGE`, `ledger.SNAPSHOT`.
- Produces:
  - `toolchains.load(text: str | None = None) -> Inventory`, which reads the
    packaged TOML by default and raises `ValueError` naming each problem;
  - `Environment(name, image, suite, snapshot, ci)` and
    `Tarball(name, url, sha256, strip, mirror)` with `Tarball.root`
    (`/opt/toolchains/<name>`); `mirror` marks a third-party copy of a
    vendor's file;
  - `Toolchain(name, environment, compiler, cpus, packages, tarball, vendor_fork, stack, link)`
    with `.family` and `.targets() -> list[measure.Target]`;
  - `Libc(name, environment, kind, cc, packages, tarball)`;
  - `Inventory(environments, tarballs, toolchains, libcs)` with
    `.toolchains_in(env)`, `.libcs_in(env)`, `.packages(env)`,
    `.tarballs_for(env)` and `.label(toolchain) -> str` (`""`,
    `"vendor fork"`, or `"vendor fork, unofficial mirror"`);
  - `uspiflash toolchains list [--environment ENV]`.

```python
"""The toolchain inventory (spec 7.2, amendment 16): what
``toolchains.toml`` lists, checked, and how to install and run each
environment. Nothing here builds a compiler: every entry is a Debian
package at a pinned snapshot or a release tarball pinned by sha256."""

from __future__ import annotations

import re
import tomllib
from dataclasses import dataclass, replace
from importlib import resources
from typing import TYPE_CHECKING, Any

from . import ledger, measure

if TYPE_CHECKING:
    from collections.abc import Mapping

#: Where tarballs are unpacked, inside a container.
OPT = "/opt/toolchains"
#: What every environment installs besides its toolchains: git (hatch-vcs
#: derives the version from it), curl and xz for tarballs, certificates.
BASE_PACKAGES = ("bash", "ca-certificates", "curl", "git", "xz-utils")
_SNAPSHOT = re.compile(r"\d{8}T\d{6}Z")
_SHA256 = re.compile(r"[0-9a-f]{64}")


@dataclass(frozen=True)
class Environment:
    """A digest-pinned Debian image and the snapshot its packages come from."""

    name: str
    image: str
    suite: str
    snapshot: str
    #: "pr": matrix.yml measures it on pull requests too; "nightly": on
    #: main and nightly only.
    ci: str


@dataclass(frozen=True)
class Tarball:
    """A release tarball, pinned by sha256."""

    name: str
    url: str
    sha256: str
    #: Leading directories to strip when unpacking.
    strip: int
    #: A third-party copy of a vendor's file (WCH's official URL answers
    #: 403): labelled "unofficial mirror" wherever its toolchains appear.
    mirror: bool = False

    @property
    def root(self) -> str:
        """Where it is unpacked in a container."""
        return f"{OPT}/{self.name}"


@dataclass(frozen=True)
class Toolchain:
    """One compiler and the CPUs the matrix measures it on."""

    name: str
    environment: str
    compiler: str
    cpus: tuple[str, ...]
    packages: tuple[str, ...] = ()
    tarball: str | None = None
    #: A vendor's fork (WCH's MounRiver GCC), labelled so everywhere.
    vendor_fork: bool = False
    #: Whether it takes -fstack-usage.
    stack: bool = True
    #: Whether a linker for its CPUs is packaged with it.
    link: bool = True

    @property
    def family(self) -> str:
        """``gcc``, ``clang`` or ``sdcc``."""
        return measure.family_of(self.compiler)

    def targets(self) -> list[measure.Target]:
        """A :class:`~uspiflash.measure.Target` per CPU."""
        out = []
        for cpu in self.cpus:
            t = measure.target(cpu, self.compiler, stack=self.stack)
            out.append(t if self.link else replace(t, link_flags=None))
        return out


@dataclass(frozen=True)
class Libc:
    """A C library the libc matrix builds against (Part D)."""

    name: str
    environment: str
    #: "linux" (the spidev tool) or "baremetal" (examples/baremetal).
    kind: str
    #: The compiler command, with the flags that select the library.
    cc: tuple[str, ...]
    packages: tuple[str, ...] = ()
    tarball: str | None = None


@dataclass(frozen=True)
class Inventory:
    """Everything ``toolchains.toml`` lists."""

    environments: Mapping[str, Environment]
    tarballs: Mapping[str, Tarball]
    toolchains: tuple[Toolchain, ...]
    libcs: tuple[Libc, ...]

    def toolchains_in(self, env: str) -> list[Toolchain]:
        """The toolchains installed in ``env``."""
        return [t for t in self.toolchains if t.environment == env]

    def libcs_in(self, env: str) -> list[Libc]:
        """The C libraries installed in ``env``."""
        return [c for c in self.libcs if c.environment == env]

    def packages(self, env: str) -> list[str]:
        """Every Debian package ``env`` installs, sorted."""
        entries = [*self.toolchains_in(env), *self.libcs_in(env)]
        return sorted({*BASE_PACKAGES, *(p for e in entries for p in e.packages)})

    def tarballs_for(self, env: str) -> list[Tarball]:
        """The tarballs ``env`` unpacks, by name."""
        entries = [*self.toolchains_in(env), *self.libcs_in(env)]
        names = sorted({e.tarball for e in entries if e.tarball})
        return [self.tarballs[n] for n in names]

    def label(self, t: Toolchain) -> str:
        """How ``t`` is labelled in every listing: ``vendor fork``, with
        ``, unofficial mirror`` when its tarball is one; else nothing."""
        if not t.vendor_fork:
            return ""
        mirror = t.tarball is not None and self.tarballs[t.tarball].mirror
        return "vendor fork, unofficial mirror" if mirror else "vendor fork"


def _environment(name: str, d: dict[str, Any], errors: list[str]) -> Environment:
    env = Environment(name, d["image"], d["suite"], d["snapshot"], d.get("ci", "pr"))
    if "@sha256:" not in env.image:
        errors.append(f"environment {name}: the image must be pinned by digest")
    if not _SNAPSHOT.fullmatch(env.snapshot):
        errors.append(f"environment {name}: snapshot must look like 20260918T000000Z")
    if env.ci not in ("pr", "nightly"):
        errors.append(f"environment {name}: ci must be pr or nightly")
    return env


def _tarball(name: str, d: dict[str, Any], errors: list[str]) -> Tarball:
    t = Tarball(
        name, d["url"], d["sha256"], int(d.get("strip", 0)), mirror=bool(d.get("mirror", False))
    )
    if not t.url.startswith("https://"):
        errors.append(f"tarball {name}: the URL must be https")
    if not _SHA256.fullmatch(t.sha256):
        errors.append(f"tarball {name}: sha256 must be 64 hex digits")
    return t


def _check_entry(
    name: str,
    env: str,
    packages: tuple[str, ...],
    tarball: str | None,
    command: str,
    inv: tuple[Mapping[str, Environment], Mapping[str, Tarball]],
    errors: list[str],
) -> None:
    environments, tarballs = inv
    if env not in environments:
        errors.append(f"{name}: no environment {env}")
    if not packages and tarball is None:
        errors.append(f"{name}: neither packages nor a tarball")
    if tarball is not None:
        if tarball not in tarballs:
            errors.append(f"{name}: no tarball {tarball}")
        elif not command.startswith(tarballs[tarball].root + "/"):
            errors.append(f"{name}: its compiler must be under {tarballs[tarball].root}")


def load(text: str | None = None) -> Inventory:
    """The inventory in ``text``, or the packaged ``toolchains.toml``.
    Raises ``ValueError`` naming every problem found."""
    if text is None:
        text = resources.files("uspiflash").joinpath("toolchains.toml").read_text("utf-8")
    data = tomllib.loads(text)
    errors: list[str] = []
    if data.get("format") != 1:
        errors.append("format must be 1")
    envs = {n: _environment(n, d, errors) for n, d in data.get("environments", {}).items()}
    tars = {n: _tarball(n, d, errors) for n, d in data.get("tarballs", {}).items()}
    seen: set[str] = set()
    chains = []
    for d in data.get("toolchains", []):
        t = Toolchain(
            name=d["name"],
            environment=d["environment"],
            compiler=d["compiler"],
            cpus=tuple(d["cpus"]),
            packages=tuple(d.get("packages", ())),
            tarball=d.get("tarball"),
            vendor_fork=bool(d.get("vendor_fork", False)),
            stack=bool(d.get("stack", True)),
            link=bool(d.get("link", True)),
        )
        if t.name in seen:
            errors.append(f"{t.name}: two toolchains of this name")
        seen.add(t.name)
        _check_entry(t.name, t.environment, t.packages, t.tarball, t.compiler, (envs, tars), errors)
        for cpu in t.cpus:
            if cpu not in measure.CPUS:
                errors.append(f"{t.name}: unknown CPU {cpu}")
                continue
            try:
                measure.target(cpu, t.compiler)
            except ValueError as e:
                errors.append(f"{t.name}: {e}")
        chains.append(t)
    libcs = []
    for d in data.get("libcs", []):
        c = Libc(
            d["name"],
            d["environment"],
            d["kind"],
            tuple(d["cc"]),
            tuple(d.get("packages", ())),
            d.get("tarball"),
        )
        if c.kind not in ("linux", "baremetal"):
            errors.append(f"{c.name}: kind must be linux or baremetal")
        _check_entry(c.name, c.environment, c.packages, c.tarball, c.cc[0], (envs, tars), errors)
        libcs.append(c)
    trixie = envs.get("trixie")
    if trixie is not None and (trixie.image, trixie.snapshot) != (ledger.IMAGE, ledger.SNAPSHOT):
        errors.append("environment trixie: must be ledger.IMAGE at ledger.SNAPSHOT")
    if errors:
        raise ValueError("toolchains.toml: " + "; ".join(errors))
    return Inventory(envs, tars, tuple(chains), tuple(libcs))


def describe(inv: Inventory, env: str | None = None) -> str:
    """``uspiflash toolchains list``: one line per toolchain and C library."""
    lines = []
    for t in inv.toolchains:
        if env is not None and t.environment != env:
            continue
        source = inv.tarballs[t.tarball].url if t.tarball else " ".join(t.packages)
        fork = f"  ({inv.label(t)})" if t.vendor_fork else ""
        lines.append(
            f"{t.name:36} {t.environment:9} {t.family:6} {','.join(t.cpus):48} {source}{fork}"
        )
    for c in inv.libcs:
        if env is not None and c.environment != env:
            continue
        source = inv.tarballs[c.tarball].url if c.tarball else " ".join(c.packages)
        lines.append(f"{c.name:36} {c.environment:9} {'libc':6} {c.kind:48} {source}")
    return "\n".join(lines) + "\n"
```

```python
def setup_script(inv: Inventory, env: str, downloads: str) -> str:
    """The shell that installs ``env`` in its image, as root: apt from the
    environment's snapshot only, then each tarball, downloaded once into
    ``downloads`` (kept between runs) and checked against its sha256."""
    e = inv.environments[env]
    q = shlex.quote
    lines = [
        "set -eu",
        "rm -f /etc/apt/sources.list.d/debian.sources",
        f'echo "{_apt_source(e)}" > /etc/apt/sources.list',
        "echo 'Acquire::Check-Valid-Until \"false\";' > /etc/apt/apt.conf.d/99snapshot",
        "apt-get update -q",
        "DEBIAN_FRONTEND=noninteractive apt-get install -y -q --no-install-recommends "
        + " ".join(q(p) for p in inv.packages(env)),
        f"mkdir -p {q(downloads)}",
    ]
    for t in inv.tarballs_for(env):
        f = f"{downloads}/{t.sha256}"
        lines += [
            f"[ -f {q(f)} ] || curl -fsSL --retry 3 -o {q(f + '.part')} {q(t.url)}",
            f"[ -f {q(f)} ] || mv {q(f + '.part')} {q(f)}",
            f'echo "{t.sha256}  {f}" | sha256sum -c -',
            f"rm -rf {t.root} && mkdir -p {t.root}",
            f"tar -xf {f} -C {t.root} --strip-components={t.strip}",
        ]
    return "\n".join(lines) + "\n"


def docker_command(
    inv: Inventory,
    env: str,
    argv: list[str],
    root: Path,
    limits: sandbox.Limits,
    uv: str,
) -> list[str]:
    """``docker run`` for ``argv`` in ``env``: the repository at /w, the
    host's uv, the sandbox's limits. The container installs the
    environment, then runs ``argv``, then gives every file back to the
    repository's owner. Everything it writes stays in the repository:
    ``tmp/toolchains/`` caches downloads, uv's cache and uv's Pythons."""
    script = "\n".join(
        [
            "trap 'chown -R \"$(stat -c %u:%g /w)\" /w' EXIT",
            setup_script(inv, env, "/w/tmp/toolchains/downloads"),
            "git config --global --add safe.directory /w",
            shlex.join(argv),
        ]
    )
    return [
        "docker",
        "run",
        "--rm",
        f"--name=uspiflash-toolchains-{env}-{os.getpid()}",
        *sandbox.docker_limits(limits),
        "-v",
        f"{root}:/w",
        "-w",
        "/w",
        "-v",
        f"{uv}:/usr/local/bin/uv:ro",
        "-e",
        "UV_PROJECT_ENVIRONMENT=/venv",
        "-e",
        "UV_CACHE_DIR=/w/tmp/toolchains/uv-cache",
        "-e",
        "UV_PYTHON_INSTALL_DIR=/w/tmp/toolchains/python",
        inv.environments[env].image,
        "bash",
        "-c",
        script,
    ]


def run(inv: Inventory, env: str, argv: list[str], root: Path) -> int:
    """Run ``argv`` in ``env`` (or in every environment with toolchains, one
    after another, for ``all``; ``{env}`` in ``argv`` becomes each name),
    with limits sized from the machine's load now. Returns the first
    non-zero exit status, else 0."""
    uv = shutil.which("uv")
    if uv is None:
        msg = "uv is not on PATH: toolchains run mounts the host's uv into the container"
        raise SystemExit(msg)
    names = [e for e in inv.environments if inv.toolchains_in(e)] if env == "all" else [env]
    status = 0
    for name in names:
        inner = [a.replace("{env}", name) for a in argv]
        cmd = docker_command(inv, name, inner, root, sandbox.current_limits(), uv)
        print(f"== {name}: {shlex.join(inner)}", flush=True)
        res = subprocess.run(cmd, check=False)
        status = status or res.returncode
    return status


def ci_matrix(inv: Inventory, *, pull_request: bool) -> dict[str, Any]:
    """matrix.yml's plan: each environment with toolchains that runs on
    this event, and the libc environment. ``pull_request`` limits it to
    the "pr" environments; matrix.yml passes it only for a pull request
    that leaves the C template and the inventory alone, so a template
    change is checked on every compiler before it merges."""

    def entry(e: Environment) -> dict[str, str]:
        return {"environment": e.name, "image": e.image, "suite": e.suite, "snapshot": e.snapshot}

    measure_envs = [
        entry(e)
        for e in inv.environments.values()
        if inv.toolchains_in(e.name) and (e.ci == "pr" or not pull_request)
    ]
    return {"measure": measure_envs, "libc": entry(inv.environments["libc"])}
```

```python
def diff_files(root: Path, files: Mapping[Path, str]) -> str:
    """A unified diff from the committed files under ``root`` to ``files``
    (empty when they match)."""
    out: list[str] = []
    for path, want in files.items():
        p = root / path
        have = p.read_text(encoding="utf-8") if p.is_file() else ""
        out += difflib.unified_diff(
            have.splitlines(keepends=True),
            want.splitlines(keepends=True),
            f"{path} (committed)",
            f"{path} (measured)",

    return "".join(out)
```

def _entry(s: Sizes) -> dict[str, int | None]:
    return {
        "text": s.text,
        "rodata": s.rodata,
        "frames": s.frames,
        "total": s.total,
        "linked": s.linked,
        "stack": s.stack.peak if s.stack else None,

def build(
    env: str,
    workdir: Path,
    inv: toolchains.Inventory | None = None,
    jobs: int | None = None,
) -> dict[str, Any]:
    """Measure every build of ``env`` at each of its CPU's configurations
    (:func:`~uspiflash.measure.configs_for`), ``jobs`` compiles at a time
    (default: the sandbox's CPU count now). Every failure is collected and
    raised together, as one ``MeasureError``."""
    inv = inv or toolchains.load()
    chains = inv.toolchains_in(env)
    cfgs = list(measure.CONFIGS)
    headers = {name: emit.render(config) for name, config in cfgs}
    work = [(tc, t) for tc in chains for t in tc.targets()]
    with ThreadPoolExecutor(max_workers=jobs or sandbox.current_limits().cpus) as pool:
        futures = {
            (tc.name, t.name, name): pool.submit(
                measure.measure, config, t, workdir / tc.name / t.name / name, header=headers[name]
            )
            for tc, t in work
            for name, config in measure.configs_for(t.name)
        }
        results: dict[tuple[str, str, str], Sizes] = {}
        errors: list[str] = []
        for key, f in futures.items():
            try:
                results[key] = f.result()
            except measure.MeasureError as e:
                errors.append(f"{key[0]} {key[1]} {key[2]}: {e}")
    if errors:
        raise measure.MeasureError("\n".join(errors))
    e = inv.environments[env]
    return {
        "format": FORMAT,
        "environment": {"name": e.name, "image": e.image, "suite": e.suite, "snapshot": e.snapshot},
        "spiflash": {"version": spiflash.__version__, "database_format": DB_FORMAT},
        "tools": {tc.name: measure.version(tc.compiler) for tc in chains},
        "toolchains": [
            {
                "name": tc.name,
                "compiler": tc.compiler,
                "family": tc.family,
                "packages": list(tc.packages),
                "tarball": inv.tarballs[tc.tarball].url if tc.tarball else None,
                "vendor_fork": tc.vendor_fork,
                "label": inv.label(tc),
            }
            for tc in chains
        ],
        "configs": [{"name": n, "options": ledger.options(c)} for n, c in cfgs],
        "builds": [
            {
                "toolchain": tc.name,
                "cpu": t.name,
                "flags": list(t.flags),
                "runtime": sorted(t.runtime),
            }

        "sizes": {
            f"{tc.name}/{t.name}": {
                n: _entry(results[(tc.name, t.name, n)]) for n, _ in measure.configs_for(t.name)
            }
            for tc, t in work
        },
    }

def files(root: Path, env: str, data: dict[str, Any]) -> dict[Path, str]:
    """The environment's file and the README, with the other environments'
    files as committed."""
    every = {**committed(root), env: data}
    return {DIR / f"{env}.json": dumps(data), README: readme(dict(sorted(every.items())))}


def tool_mismatches(data: dict[str, Any]) -> list[str]:
    """How the installed compilers differ from the ones that made ``data``."""
    out = []
    for tc in data["toolchains"]:
        want = data["tools"][tc["name"]]
        try:
            have = measure.version(tc["compiler"])
        except measure.MeasureError as e:
            have = f"missing ({e})"
        if have != want:
            out.append(f"{tc['name']}: the matrix has {want!r}, installed is {have!r}")
    return out


def measure_fresh(root: Path, env: str) -> dict[str, Any]:
    """:func:`build`, with scratch files under ``root/tmp``, removed after."""
    scratch = root / "tmp"
    scratch.mkdir(parents=True, exist_ok=True)
    with tempfile.TemporaryDirectory(prefix="uspiflash-matrix-", dir=scratch) as d:
        return build(env, Path(d))


def _git_json(root: Path, ref: str, path: Path) -> dict[str, Any] | None:
    res = subprocess.run(
        ["git", "-C", str(root), "show", f"{ref}:{path}"],
        capture_output=True,
        text=True,
        check=False,
    )
    if res.returncode != 0:
        return None
    data: dict[str, Any] = json.loads(res.stdout)
    return data


def _totals(data: dict[str, Any] | None, *, matrix_file: bool) -> dict[tuple[str, str], int]:
    if data is None:
        return {}
    if matrix_file:
        return {
            (build, cfg): e["total"]
            for build, per in data["sizes"].items()
            for cfg, e in per.items()
        }
    return {(t, cfg): e["total"] for t, per in data["sizes"].items() for cfg, e in per.items()}


def compare(root: Path, ref: str) -> str:
    """A Markdown table of every total that differs between the size files
    committed at git ``ref`` and the working tree's: the reference ledger
    first, then each matrix file, SPI NOR configurations first."""
    pairs = [(ledger.LEDGER, False)]
    names = {p.name for p in (root / DIR).glob("*.json")}
    pairs += [(DIR / n, True) for n in sorted(names)]
    rows = []
    for path, is_matrix in pairs:
        old = _totals(_git_json(root, ref, path), matrix_file=is_matrix)
        p = root / path
        here = json.loads(p.read_text(encoding="utf-8")) if p.is_file() else None
        new = _totals(here, matrix_file=is_matrix)
        for key in sorted(old.keys() & new.keys(), key=lambda k: (not k[1].endswith(":nor"), k)):
            if old[key] != new[key]:
                delta = new[key] - old[key]
                rows.append(
                    f"| `{key[0]}` | `{key[1]}` | {old[key]:,} | {new[key]:,} | "
                    f"{delta:+,} ({delta / old[key]:+.1%}) |"
                )
    if not rows:
        return f"No size changes against {ref}.\n"
    head = [
        f"Size changes against `{ref}` (bytes of flash, code and tables):",
        "",
        "| Build | Configuration | Before | After | Change |",
        "|---|---|--:|--:|--:|",
    ]
    return "\n".join([*head, *rows]) + "\n"

```yaml
name: Size matrix

# The compiler matrix (spec 7.2-7.3, M2): every toolchain in
# src/uspiflash/toolchains.toml builds the generated library for every CPU
# it targets, and sizes/matrix/<environment>.json must match. Each
# environment runs in its own digest-pinned Debian image, with packages from
# its snapshot.debian.org timestamp.
#
# Its own workflow, never part of deb.yml ("Debian packages"):
# publish-pypi.yml follows deb.yml alone, so a slow or failing matrix job
# never blocks publishing. Pull requests run the "pr" environments; main,
# the nightly run and a manual dispatch run all of them.
#
# A stale file fails its job, which uploads the regenerated files as an
# artifact. A nightly failure also opens (or comments on) an issue
# labelled size-matrix, so the ledger never goes stale silently.

on:
  pull_request:
    paths:
      - "src/uspiflash/**"
      - "sizes/**"
      - "tests/test_portability.py"
      - "tests/test_sdcc.py"
      - "pyproject.toml"
      - "uv.lock"
      - ".github/workflows/matrix.yml"
  push:
    branches: [main]
  schedule:
    - cron: "41 3 * * *"
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: matrix-${{ github.ref }}
  cancel-in-progress: ${{ github.event_name == 'pull_request' }}

jobs:
  plan:
    runs-on: ubuntu-24.04
    timeout-minutes: 10
    outputs:
      measure: ${{ steps.plan.outputs.measure }}
      libc: ${{ steps.plan.outputs.libc }}
    steps:
      - uses: actions/checkout@v5
        with:
          fetch-depth: 0
      - uses: astral-sh/setup-uv@v7
      # A pull request measures the "pr" environments, unless it touches
      # the C template or the inventory: then every environment, WCH's and
      # the tarballs' included.
      - id: plan
        env:
          EVENT: ${{ github.event_name }}
          BASE: ${{ github.base_ref }}
        run: |
          PR=""
          if [ "$EVENT" = pull_request ] && git diff --quiet "origin/$BASE...HEAD" -- \
              src/uspiflash/templates src/uspiflash/toolchains.toml; then
            PR=--pull-request
          fi
          uv run uspiflash toolchains ci-matrix $PR > tmp-plan.json
          echo "measure=$(jq -c .measure tmp-plan.json)" >> "$GITHUB_OUTPUT"
          echo "libc=$(jq -c .libc tmp-plan.json)" >> "$GITHUB_OUTPUT"
          rm tmp-plan.json

  measure:
    name: measure (${{ matrix.environment }})
    needs: plan
    runs-on: ubuntu-24.04
    timeout-minutes: 90
    container: ${{ matrix.image }}
    strategy:
      fail-fast: false
      matrix:
        include: ${{ fromJSON(needs.plan.outputs.measure) }}
    steps:
      # git before checkout (hatch-vcs derives the version from history),
      # from the environment's own snapshot.
      - name: git and certificates from the pinned snapshot
        run: |
          rm -f /etc/apt/sources.list.d/debian.sources
          echo "deb http://snapshot.debian.org/archive/debian/${{ matrix.snapshot }}/ ${{ matrix.suite }} main" > /etc/apt/sources.list
          echo 'Acquire::Check-Valid-Until "false";' > /etc/apt/apt.conf.d/99snapshot
          apt-get update -q
          apt-get install -y -q --no-install-recommends ca-certificates git
      - uses: actions/checkout@v5
        with:
          fetch-depth: 0
      - name: Trust the checkout
        run: git config --global --add safe.directory "$GITHUB_WORKSPACE"
      - uses: astral-sh/setup-uv@v7
      # Tarballs are cached by their sha256s (the inventory's hash is the key).
      - uses: actions/cache@v6
        with:
          path: tmp/toolchains/downloads
          key: toolchains-${{ matrix.environment }}-${{ hashFiles('src/uspiflash/toolchains.toml') }}
      - name: Install the environment
        run: |
          mkdir -p tmp
          uv run uspiflash toolchains script ${{ matrix.environment }} --downloads "$PWD/tmp/toolchains/downloads" > tmp/setup.sh
          bash tmp/setup.sh
      - name: Check the size matrix
        run: uv run uspiflash measure --matrix ${{ matrix.environment }} --check
      - name: Portability and SDCC tests
        if: matrix.environment == 'trixie'
        run: >-
          uv run pytest tests/test_portability.py tests/test_sdcc.py tests/test_cpus.py -v -ra
          -o addopts="--basetemp=tmp/pytest"
      - name: Regenerate on failure
        if: failure()
        run: uv run uspiflash measure --matrix ${{ matrix.environment }} --write
      - uses: actions/upload-artifact@v7
        if: failure()
        with:
          name: sizes-matrix-${{ matrix.environment }}
          path: sizes/matrix/

  size-diff:
    if: github.event_name == 'pull_request' && github.event.pull_request.head.repo.full_name == github.repository
    runs-on: ubuntu-24.04
    timeout-minutes: 10
    permissions:
      contents: read
      pull-requests: write
    steps:
      - uses: actions/checkout@v5
        with:
          fetch-depth: 0
      - uses: astral-sh/setup-uv@v7
      - name: Compare the committed size files with the base branch
        env:
          GH_TOKEN: ${{ github.token }}
          PR: ${{ github.event.pull_request.number }}
          BASE: origin/${{ github.base_ref }}
        run: |
          mkdir -p tmp
          uv run uspiflash measure --compare "$BASE" > tmp/size-diff.md
          cat tmp/size-diff.md
          gh pr comment "$PR" --edit-last --body-file tmp/size-diff.md \
            || gh pr comment "$PR" --body-file tmp/size-diff.md

  # Part D adds libc and host-musl to needs, so their nightly failures
  # open the issue too.
  stale:
    needs: [measure]
    if: failure() && github.event_name == 'schedule'
    runs-on: ubuntu-24.04
    timeout-minutes: 5
    permissions:
      issues: write
    steps:
      - name: Open or update the issue
        env:
          GH_TOKEN: ${{ github.token }}
          RUN: ${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}

          body="The nightly size matrix failed: $RUN (the regenerated files are the run's artifacts)."
          n=$(gh issue list --repo "$GITHUB_REPOSITORY" --label size-matrix --state open --json number --jq '.[0].number')

          else

              --description "The size matrix check failed" || true
            gh issue create --repo "$GITHUB_REPOSITORY" --label size-matrix \
              --title "Size matrix: nightly check failed" --body "$body"
          fi
```

```python
from uspiflash.sandbox import current_limits

ROOT = Path(__file__).resolve().parent.parent.parent
REPOSITORY = "ghcr.io/mithro/uspiflash-toolchains"
DOWNLOADS = "/var/cache/uspiflash-downloads"


def dockerfile(inv: toolchains.Inventory, env: str) -> str:
    """The Dockerfile: the environment's image, then its setup script."""
    return (
        f"FROM {inv.environments[env].image}\n"
        "COPY setup.sh /setup.sh\n"
        f"RUN bash /setup.sh && rm -rf /setup.sh {DOWNLOADS} /var/lib/apt/lists/*\n"
    )


def image_tag(inv: toolchains.Inventory, env: str) -> str:
    """``REPOSITORY:<env>-<12 hex digits of the setup script's sha256>``."""
    script = toolchains.setup_script(inv, env, DOWNLOADS)
    return f"{REPOSITORY}:{env}-{hashlib.sha256(script.encode()).hexdigest()[:12]}"


def main(argv: list[str] | None = None) -> int:
    """Build (and push) the image."""
    ap = argparse.ArgumentParser(prog="build_images.py")
    ap.add_argument("environment")
    ap.add_argument("--push", action="store_true", help="push to ghcr.io (docker login first)")
    args = ap.parse_args(argv)
    inv = toolchains.load()
    ctx = ROOT / "tmp" / "images" / args.environment
    ctx.mkdir(parents=True, exist_ok=True)
    (ctx / "setup.sh").write_text(toolchains.setup_script(inv, args.environment, DOWNLOADS))
    (ctx / "Dockerfile").write_text(dockerfile(inv, args.environment))
    tag = image_tag(inv, args.environment)
    limits = current_limits()
    build = [
        "docker",
        "build",
        f"--memory={limits.memory_bytes}",
        f"--cpu-quota={limits.cpus * 100000}",
        "-t",
        tag,
        str(ctx),
    ]
    if subprocess.run(build, check=False).returncode != 0:
        return 1
    print(tag, flush=True)
    if args.push and subprocess.run(["docker", "push", tag], check=False).returncode != 0:
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

(`docker build`'s `--memory` and `--cpu-quota` limit only the legacy
builder's RUN steps. BuildKit ignores them. `DOCKER_BUILDKIT=0` selects
the legacy builder, which Docker has deprecated (since Docker 23 it
prints a deprecation warning) and may remove. The images are optional
tooling, so if it goes, their builds run without the limits.)

`tools/images/README.md` covers:
- what the images are, and that they are optional;
- the tag rule;
- the build command and the `--push` command (after
  `echo $TOKEN | docker login ghcr.io -u <user> --password-stdin`);
- that `DOCKER_BUILDKIT=0` makes the sandbox's limits apply to the build,
  and that it selects Docker's deprecated legacy builder, which may go.
