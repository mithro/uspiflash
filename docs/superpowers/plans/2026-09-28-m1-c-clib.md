## Rules for the C template (all tasks)

- **Language:** C99, and only what SDCC accepts.
  - No `//` comments (SDCC accepts them, but the house style is `/* */`).
  - No VLAs, no designated initialisers, no compound literals, no struct
    passing or returning by value, no `inline`, no `_Bool`.
- **Types:** `uint8_t` for everything that fits, `uint16_t` for entry
  indices and offsets, `uint32_t` only for sizes.
- **Tables:** every table is `static const USF_ROM uint8_t name[]`.
  `USF_ROM` defaults to nothing; SDCC users define it as `__code`.
- **No compiler helpers:** no `*`, `/` or `%` by a non-constant. Multiplying
  by a constant is fine: compilers turn it into shifts and adds.
- **Guards:** every static helper sits inside the `#if` of the fields that
  use it, so that no level warns about unused functions or variables.
- **Symbols:** public symbols are `usf_*`, internal ones `usf__*`, macros
  `USF_*`. `emit.py` renames all three for `--prefix`.
- **Comments:** every public declaration has a comment saying what it
  returns and what "unknown" looks like.
- **No libc names:** never name a parameter, local or struct field after a
  libc function (`putc`, `getc`, `index`, `abs`, ...). A libc may define
  them as function-like macros, which would expand inside the library
  whenever a user includes `<stdio.h>` first. The character sink is
  therefore always `sink`.

"""Shared fixtures: generating headers and compiling C with the strict flags."""

from __future__ import annotations

import shutil
import subprocess
from typing import TYPE_CHECKING

import pytest

from uspiflash import emit
from uspiflash.provenance import Config

if TYPE_CHECKING:
    from collections.abc import Sequence
    from pathlib import Path

CFLAGS = ["-std=c99", "-Wall", "-Wextra", "-Wpedantic", "-Wundef", "-Werror"]


@pytest.fixture(scope="session")
def compilers() -> list[str]:
    found = [cc for cc in ("gcc", "clang") if shutil.which(cc)]
    if not found:
        pytest.skip("no C compiler")
    return found


def generate(tmp: Path, config: Config) -> Path:
    """Write the header ``config`` describes into ``tmp``; return its path."""
    path = tmp / config.filename
    path.write_text(emit.render(config), encoding="ascii")
    return path


def compile_c(cc: str, sources: Sequence[str | Path], out: Path, extra: Sequence[str] = ()) -> None:
    """Compile with the strict flags; any diagnostic fails the test."""
    cmd = [cc, *CFLAGS, *extra, *(str(s) for s in sources), "-o", str(out)]
    res = subprocess.run(cmd, capture_output=True, text=True, check=False)
    assert res.returncode == 0 and not res.stderr, f"{' '.join(cmd)}\n{res.stderr}"

```python
"""Small helpers that write C source text."""

from __future__ import annotations


def byte_array(name: str, data: bytes, per_line: int = 16) -> str:
    """``static const USF_ROM uint8_t name[] = {...};`` (never empty: C forbids
    a zero-length initialiser, so an empty table holds one unused 0)."""
    body = data or b"\0"
    lines = [
        "    " + ", ".join(f"0x{b:02x}" for b in body[i : i + per_line]) + ","
        for i in range(0, len(body), per_line)
    ]
    size = f"/* {len(data)} bytes */" if data else "/* empty: nothing needs it */"
    return f"static const USF_ROM uint8_t usf_{name}[] = {{ {size}\n" + "\n".join(lines) + "\n};\n"


def defines(values: dict[str, int]) -> str:
    """``#define USF_<NAME> <value>`` lines, sorted by name."""
    return "".join(f"#define USF_{k} {v}\n" for k, v in sorted(values.items()))
```

```python
"""The comment every generated file starts with: what made it, from what,
and how to make it again. It embeds the configuration as one JSON line that
``uspiflash check`` reads back."""

from __future__ import annotations

import hashlib
import json
import re
from dataclasses import dataclass
from typing import Any

import spiflash

from . import __version__
from .levels import LEVELS, Selection

CONFIG_RE = re.compile(r"^ \* uspiflash-config: (\{.*\})$", re.MULTILINE)


@dataclass(frozen=True)
class Config:
    """Everything that determines a generated file's content (with the
    installed uspiflash and spiflash versions)."""

    selection: Selection
    prefix: str = "usf"
    filename: str = "uspiflash.h"

    def to_json(self) -> dict[str, Any]:
        """Plain JSON; key order is fixed so the digest is stable."""
        return {"prefix": self.prefix, "filename": self.filename,
                "selection": self.selection.to_json()}

    @classmethod
    def from_json(cls, d: dict[str, Any]) -> Config:
        """The inverse of :meth:`to_json`."""
        return cls(Selection.from_json(d["selection"]), d["prefix"], d["filename"])

    def digest(self) -> str:
        """SHA-256 of the canonical JSON."""
        text = json.dumps(self.to_json(), sort_keys=True, separators=(",", ":"))
        return hashlib.sha256(text.encode()).hexdigest()

    def args(self) -> list[str]:
        """The ``uspiflash generate`` arguments that reproduce this config."""
        sel = self.selection
        out = ["-o", self.filename, "--prefix", self.prefix, "--level", sel.level]
        base = LEVELS[sel.level]
        out += [a for f in sorted(sel.fields - base) for a in ("--with", str(f))]
        out += [a for f in sorted(base - sel.fields) for a in ("--without", str(f))]
        c = sel.chips
        out += [a for m in c.manufacturers for a in ("--manufacturer", m)]
        out += [a for i in c.ids for a in ("--id", i.hex())]
        out += [a for t in c.types for a in ("--type", t.value)]
        out += [a for f in c.families for a in ("--family", f.value)]
        if c.min_size is not None:
            out += ["--min-size", str(c.min_size)]
        if c.max_size is not None:
            out += ["--max-size", str(c.max_size)]
        return out


def header(config: Config) -> str:
    """The provenance comment."""
    args = " ".join(_quote(a) for a in config.args())
    src = spiflash.sources()
    lines = [
        f"{config.filename}: detect SPI flash chips and describe them.",
        "",
        f"Generated by uspiflash {__version__} from spiflash {spiflash.__version__}.",
        "Do not edit: regenerate it instead.",
        "",
        "The spiflash data was extracted from these upstream commits:",
        *(f"  {name} {info.commit} ({info.date:%Y-%m-%d}, {info.license})"
          for name, info in sorted(src.items())),
        "",
        "Regenerate (byte-identical) with:",
        f"  uvx --from 'uspiflash=={__version__}' --with 'spiflash=={spiflash.__version__}' \\",
        f"      uspiflash generate {args}",
        f"Check a committed copy is current with: uspiflash check {config.filename}",
        "",
        f"Configuration (sha256 {config.digest()}):",
        f"uspiflash-config: {json.dumps(config.to_json(), sort_keys=True)}",
        "",
        "Licence: Apache-2.0. The data is from spiflash (Apache-2.0), itself",
        "extracted from the flash tables of Linux, U-Boot, flashrom, flashprog,",
        "OpenOCD and openFPGALoader; see https://github.com/mithro/spiflash.",
        "",
        "Documentation: https://uspiflash.readthedocs.io/",
    ]
    body = "\n".join(f" * {line}".rstrip() for line in lines)
    if "*/" in body:
        msg = "provenance would end the comment early"
        raise ValueError(msg)
    return f"/*\n{body}\n */\n"


def _quote(arg: str) -> str:
    return arg if re.fullmatch(r"[\w.,:/=+-]+", arg) else "'" + arg.replace("'", "'\\''") + "'"
```

```python
"""Render the C template with a layout's tables and a provenance header."""

from __future__ import annotations

import re
from importlib import resources
from typing import TYPE_CHECKING

from . import layout
from .cgen import byte_array, defines
from .model import Snapshot
from .provenance import header

if TYPE_CHECKING:
    from .provenance import Config

_IDENT = re.compile(r"[A-Za-z_][A-Za-z0-9_]*")


def _guard(filename: str) -> str:
    g = re.sub(r"\W", "_", filename).upper()
    return ("_" + g if g[0].isdigit() else g) + "_INCLUDED"


def render(config: Config, snapshot: Snapshot | None = None) -> str:
    """The generated file for ``config``."""
    if not _IDENT.fullmatch(config.prefix):
        msg = f"prefix {config.prefix!r} is not a C identifier"
        raise ValueError(msg)
    snap = snapshot or Snapshot.build(config.selection.chips.apply(Snapshot.build().database))
    lay = layout.build(snap, config.selection)
    template = resources.files("uspiflash").joinpath("templates/uspiflash.h.in").read_text("ascii")
    text = (
        template.replace("@@PROVENANCE@@", header(config))
        .replace("@@GUARD@@", _guard(config.filename))
        .replace("@@CONFIG@@", defines(lay.defines) + _constants())
        .replace("@@TABLES@@", "".join(byte_array(n, d) for n, d in sorted(lay.tables.items())))
    )
    if config.prefix != "usf":
        text = re.sub(r"\busf_", config.prefix + "_", text)
        text = re.sub(r"\bUSF_", config.prefix.upper() + "_", text)
    return text
```

`_constants()` returns the `#define`s generated from spiflash's own
vocabularies, so the C names can never drift from the data:
- `USF_FAMILY_<NAME>` (index in `FAMILIES`)
- `USF_TYPE_NOR 0`, `USF_TYPE_NAND 1`
- `USF_FEATURE_<NAME>` (index in `layout.FEATURES`; name upper-cased, with
  `4BYTE_ADDR` spelled `FOUR_BYTE_ADDR`, as spiflash's enum member names are)
- `USF_SOURCE_<NAME>` (index in `layout.SOURCES`, member names)
- `USF_KIND_<NAME>` (index in `layout.KINDS`)
- `USF_PROTO_<n_n_n>` (index in `layout.PROTOCOLS`, `-` → `_`, `D` kept)
- `USF_OP_<NAME>` (index in `layout.ALL_OPS`)

Use the enum *member* names for everything, so `USF_FEATURE_QUAD_READ` and
`USF_SOURCE_UBOOT` read naturally.

Building a snapshot of a filtered database: `ChipFilter.apply` returns a
`Database`; `Snapshot.build(that)` numbers entries within the subset.

#ifndef @@GUARD@@
#define @@GUARD@@

#ifdef __cplusplus
extern "C" {
#endif

/*
 * Types. <stdint.h> is a freestanding header (no libc needed); define
 * USF_NO_STDINT and provide uint8_t, uint16_t and uint32_t yourself if your
 * environment has no <stdint.h> at all.
 */
#ifndef USF_NO_STDINT
#include <stdint.h>
#endif

/* ------------------------------------------------------------------------
 * What this file was generated with (see the comment at the top): which
 * fields are present (USF_HAVE_*), table geometry, and the constants that
 * name spiflash's families, features, sources, operation kinds, protocols
 * and operations.
 * ---------------------------------------------------------------------- */
@@CONFIG@@

/*
 * Where the tables live. Nothing by default; define as __code for SDCC on
 * the 8051 so they stay in program memory.
 */
#ifndef USF_ROM
#define USF_ROM
#endif

/* The longest id usf_id() writes (after 0x7f continuation codes). */
#define USF_ID_MAX 7

/* A character sink: every string leaves the library through one of these,
 * one character at a time, so no buffer is ever needed. */
typedef void (*usf_putc_fn)(void *ctx, char ch);

/* One answer: which entry (a chip id, or a variant of it narrowed by its
 * extended id) and the chip id it belongs to. Filled by usf_lookup() and
 * usf_probe(); four bytes of RAM. */
typedef struct usf_chip {
    uint16_t entry;
    uint16_t base;
} usf_chip;

/*
 * Look up the bytes a chip answered to an id command of `family`
 * (USF_FAMILY_JEDEC for 0x9F, or a legacy family). Leading 0x7f
 * continuation codes are skipped; bytes after the id narrow the answer to
 * the variants their extended id agrees with. Writes up to two answers to
 * `out` (a NOR chip, then a NAND chip) and returns how many: 0, 1 or 2.
 * Exactly what spiflash's `lookup(data, method=family)` returns.
 */
uint8_t usf_lookup(uint8_t family, const uint8_t *id, uint8_t len, usf_chip *out);

/* The chip's id family (USF_FAMILY_*), type (USF_TYPE_*) and JEP106 bank
 * (how many 0x7f codes precede its manufacturer byte). */
uint8_t usf_family(const usf_chip *c);
uint8_t usf_type(const usf_chip *c);
uint8_t usf_bank(const usf_chip *c);
/* Writes the chip's id bytes (without continuation codes) to `buf`, which
 * must hold USF_ID_MAX bytes; returns how many. */
uint8_t usf_id(const usf_chip *c, uint8_t *buf);

/* Size in bytes; 0 when unknown. */
uint32_t usf_size(const usf_chip *c);
#endif
#if USF_HAVE_PAGE_SIZE
/* Program page size in bytes; 0 when unknown. */
uint32_t usf_page_size(const usf_chip *c);
#endif
#if USF_HAVE_SECTOR_SIZE
/* Sector (usually 64 KiB erase block) size in bytes; 0 when unknown. */
uint32_t usf_sector_size(const usf_chip *c);
#endif
#if USF_HAVE_VOLTAGE
/* Supply range in millivolts; returns 0 (and writes nothing) when unknown. */

#endif

uint8_t usf_has_feature(const usf_chip *c, uint8_t feature);
#endif

/* One operation the chip supports, and the transaction it is on the bus. */
typedef struct usf_op {
    uint8_t id;            /* USF_OP_* */
    uint8_t opcode;        /* the command byte */
    uint8_t kind;          /* USF_KIND_* */
    uint8_t protocol;      /* USF_PROTO_*: lines for command-address-data */

    uint8_t dummy_clocks;  /* 0xFF: set in the part's registers or its SFDP */

    uint8_t sources;       /* bit USF_SOURCE_*: who vouches (0 without sources) */
} usf_op;
/* How many operations the chip has (in spiflash's order: id, read,

uint8_t usf_op_count(const usf_chip *c);

#endif

#endif

/* The ids-table record (header byte, then id bytes) of chip id `base`. */
static usf__p usf__idrec(uint16_t base)
{
    usf__p p = usf_ids;

        p += 1 + (*p & 7);

#if USF_HAVE_EXT
/* The entry for chip id `base` given the `xl` bytes read after its id: the
 * mask of its extended ids that agree with them selects a variant. */
static uint16_t usf__narrow(uint16_t base, const uint8_t *x, uint8_t xl)
{
    usf_chip b;
    usf__p p;
    uint16_t off, mask = 0, bit = 1;
    uint8_t k, n, l, i;

    b.entry = base;
    off = usf__u16(usf__row(&b) + USF_E_EXT);
    if (!xl || off == 0xFFFF)
        return base;
    p = usf_ext + off;
    for (k = *p++; k; k--, bit = (uint16_t)(bit << 1)) {
        l = *p++;
        n = l < xl ? l : xl;
        for (i = 0; i < n && p[i] == x[i]; i++)
            ;
        if (i == n)
            mask |= bit;
        p += l;
    }
    for (k = *p++; k; k--, p += 4)
        if (usf__u16(p) == mask)
            return usf__u16(p + 2);
    return base;
}
#endif

uint8_t usf_lookup(uint8_t family, const uint8_t *id, uint8_t len, usf_chip *out)
{
    usf__p p = usf_ids;
    uint16_t e = 0, best[2] = {0, 0};
    uint8_t best_len[2] = {0, 0};
    uint8_t h, l, i, t, n = 0;

    while (len > 1 && *id == 0x7f) {
        id++;
        len--;
    }
    for (; (h = *p) != 0; p += 1 + l, e++) {
        l = h & 7;
        t = (h >> 3) & 1;
        if ((h >> 4) != family || l > len || l <= best_len[t])
            continue;
        for (i = 0; i < l && p[1 + i] == id[i]; i++)
            ;
        if (i == l) {
            best_len[t] = l;
            best[t] = e;
        }
    }
    for (t = 0; t < 2; t++) {
        if (!best_len[t])
            continue;
        out[n].base = best[t];
#if USF_HAVE_EXT
        out[n].entry = usf__narrow(best[t], id + best_len[t], (uint8_t)(len - best_len[t]));
#else
        out[n].entry = best[t];
#endif
        n++;
    }
    return n;
}

}

    """Every chip id, every extended-id probe, and random near-misses."""

        f = snap.entries[i].flash

        out.append((fam, b"\x7f" * f.bank + f.id))

        out.append((rng.randrange(len(FAMILIES)), bytes(rng.randrange(256) for _ in range(rng.randint(1, 6)))))
    return out

def _names_header() -> str:
    def arr(name: str, items: Sequence[object]) -> str:
        return f"static const char *const {name}[] = {{" + ", ".join(f'"{i}"' for i in items) + "};\n"

    return (
        arr("FAMILY_NAMES", [f.value for f in FAMILIES])
        + arr("KIND_NAMES", [k.value for k in layout.KINDS])
        + arr("PROTO_NAMES", layout.PROTOCOLS)
        + arr("FEATURE_NAMES", [f.value for f in layout.FEATURES])
        + arr("SOURCE_NAMES", [s.value for s in layout.SOURCES])
        + arr("OP_NAMES", layout.ALL_OPS)
    )


@dataclass
class Harness:
    """A compiled harness binary for one generated header."""

    binary: Path

    @classmethod
    def build(cls, tmp: Path, config: Config, cc: str) -> Harness:
        tmp.mkdir(parents=True, exist_ok=True)
        generate(tmp, config)
        (tmp / "harness_names.h").write_text(_names_header(), encoding="ascii")
        binary = tmp / "harness"
        compile_c(cc, [HERE / "c" / "harness.c"], binary,
                  [*SANITIZE, f"-I{tmp}", f"-I{HERE / 'c'}",
                   f"-DUSF_HEADER=\"{config.filename}\""])
        return cls(binary)

        res = subprocess.run(
            [str(self.binary)], input="".join(c + "\n" for c in commands),
            capture_output=True, text=True, check=False, timeout=600,

`spiflash id`'s text (`spiflash/cli.py`, `describe()` and `opcode_table()`),
which the printer must reproduce character for character:

```
<ident>  <manufacturer or ?>  <names joined by ", ">  (<type>)
    size <human>[, page <human>][, sector <human>][, <v>-<v> V]
    features: <sorted features joined by " ">          (only if any)
    sources disagree on <attr>: <v> (<srcs>); <v> (<srcs>)   (per conflict, attr order size, page_size, sector_size, voltage)
    from: <sources joined by ", ">
    opcodes:                                           (with --opcodes)
    0x<oo>  <NAME padded to the longest name>  <description>  [<sources>]
```

- `<ident>` is the `jedec_id` (`7f` × bank, then the id in hex) for the
  JEDEC family, else `<family>:<id hex>`.
- `human(n)` is `n >> 30` + ` GiB` if `n` is a whole number of GiB; else
  MiB, else KiB (`n >= scale and n % scale == 0`); else `n` + ` B`. An
  unknown size prints `?`.
- A voltage prints `%g` of volts: `2700` → `2.7`, `1650` → `1.65`,
  `3000` → `3`.
- A conflict value prints as the int, or `(2700, 3600)` for a voltage.
- A chip with no operations prints
  `    no opcodes known (the sources describe none for this part)`.
- Answers are separated by one blank line.
- At `describe` (no SOURCES/CONFLICTS), the `from:` and
  `sources disagree` lines and the `  [..]` suffixes are absent;
  `oracle.text` removes them from spiflash's output the same way.

def text(db: Database, family: IdFamily, data: bytes, sel: Selection, *, opcodes: bool) -> str:
    """What `spiflash id <data> --method <family> [--opcodes]` prints, less
    what ``sel`` leaves out."""
    blocks = []
    for f in db.lookup(data, method=family):
        lines = []
        for line in describe(f, opcodes=opcodes).split("\n"):
            if line.startswith("    from: ") and not sel.has(Field.SOURCES):
                continue
            if line.startswith("    sources disagree on ") and not sel.has(Field.CONFLICTS):
                continue
            if line.startswith("    0x") and not sel.has(Field.SOURCES):
                line = line[: line.rindex("  [")]
            lines.append(line)
        blocks.append("\n".join(lines))
    return "\n\n".join(blocks) + "\n"

static void usf__human(uint32_t n, usf_putc_fn sink, void *ctx)

        mv = (uint16_t)(mv - 1000);
        whole++;
    }
    usf__dec(whole, sink, ctx);
    if (mv)
        sink(ctx, '.');
    while (mv) {           /* hundreds, tens, units, stopping at trailing zeros */
        for (d = '0'; mv >= 100; d++)

        mv = (uint16_t)(mv * 10);

/* A tiny writer for Python's json.dumps(indent=1) layout. */
typedef struct usf__json {
    usf_putc_fn sink;
    void *ctx;
    uint8_t depth;
    uint8_t first;
} usf__json;

static void usf__jnl(usf__json *j)
{
    uint8_t i;
    j->sink(j->ctx, '\n');
    for (i = 0; i < j->depth; i++)
        j->sink(j->ctx, ' ');
}

static void usf__jitem(usf__json *j)
{
    if (!j->first)
        j->sink(j->ctx, ',');
    usf__jnl(j);
    j->first = 0;
}

static void usf__jopen(usf__json *j, char ch)
{
    j->sink(j->ctx, ch);
    j->depth++;
    j->first = 1;
}

static void usf__jclose(usf__json *j, char ch)
{
    j->depth--;
    if (!j->first)
        usf__jnl(j);
    j->sink(j->ctx, ch);
    j->first = 0;
}

static void usf__jkey(usf__json *j, const char *key)
{
    usf__jitem(j);
    j->sink(j->ctx, '"');
    usf__lit(key, j->sink, j->ctx);

   * `rxlen` bytes, with chip select held for the whole transaction.
   * Define USF_XFER(bus, tx, txlen, rx, rxlen) yourself to inline your SPI
   * driver; otherwise a struct usf_bus holding a function pointer is used. */
  #ifndef USF_XFER
  typedef void (*usf_xfer_fn)(void *ctx, const uint8_t *tx, uint8_t txlen, uint8_t *rx, uint8_t rxlen);
  struct usf_bus { usf_xfer_fn xfer; void *ctx; };
  #define USF_XFER(bus, tx, txlen, rx, rxlen) ((bus)->xfer((bus)->ctx, (tx), (txlen), (rx), (rxlen)))
  #endif
  typedef struct usf_bus usf_bus;

  typedef struct usf_probe_result {
      uint8_t family;              /* the command that answered (USF_FAMILY_*) */
      uint8_t len;                 /* bytes in id[]; 0 if the bus stayed 0x00/0xFF */
      uint8_t id[USF_RDID_LEN];    /* what the chip sent */
      uint8_t count;               /* answers in chip[] (0: not in the database) */
      usf_chip chip[2];
  } usf_probe_result;

  /* Identify the chip; returns r->count. Sends only id commands (RES,
   * RDID, REMS, AT25F RDID, M95 RDID), so it cannot change the chip. */
  uint8_t usf_probe(usf_bus *bus, usf_probe_result *r);

    `void sim_parse(struct sim *s, const char *spec)` and
    `void sim_xfer(void *ctx, const uint8_t *tx, uint8_t txlen, uint8_t *rx, uint8_t rxlen)`.
    Each transfer is logged to a caller-supplied FILE as
    `> <tx hex> < <rx hex>`.

**The sim spec:** space-separated `CC/N=HEX` items. The chip answers `HEX`
(then `fill`) to a transaction whose first byte is `CC` and which sends
`N` bytes. Also `fill=FF`, and `sfdp=HEX` (the SFDP image `0x5A` reads
from; part D). Example: a Winbond W25Q128 is `9f/1=ef4018 fill=ff`.

**Probe algorithm** (the M1 index's decision 5). `usf_probe`:
1. `ab 00 00 00` → keep 2 bytes `res[2]` (a local).
2. `9f` → `USF_RDID_LEN` bytes into `r->id`. If not all-0x00/all-0xFF:
   - look up as JEDEC; return if found;
   - remember the id (copy it to a local `rdid[USF_RDID_LEN]` and set
     `saved`).
3. `9f 00` → `USF_RDID_LEN - 1` bytes; if not blank, look up as JEDEC;
   return if found.
4. `90 00 00 00` → 2 bytes; if not blank, look up as REMS; return if found.
5. If `res` isn't blank: look up `res` (2 bytes) as RES2, then `res[0]`
   (1 byte) as RES1; return the first found (with `r->id`/`len`/`family`
   set to what matched).
6. `15` → 2 bytes as AT25F; `83 00 00` → 3 bytes as ST95. Return if found.
7. Nothing found: if `saved`, restore `rdid` into `r` as the JEDEC answer
   with `count = 0` (an unknown chip); else `len = 0` (the bus is blank).
   Return 0.

"Blank" means every byte is 0x00, or every byte is 0xFF.

/* A simulated SPI flash for the tests: answers id commands from a table
 * (a "spec", see tests/test_probe.py) and RDSFDP (0x5A) from an SFDP image,
 * logging every transaction. Shared by tests/c/harness.c and
 * tests/linux/fake_spidev.c. */
#ifndef SIMCHIP_H
#define SIMCHIP_H

#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct sim_reply {
    uint8_t cmd, txlen, len;
    uint8_t data[16];
};

struct sim {
    struct sim_reply replies[8];
    unsigned nreplies;
    uint8_t fill;
    uint8_t sfdp[4096];
    unsigned sfdp_len;
    FILE *log;                    /* where to log transactions, or NULL */
};

static unsigned sim_hex(const char *s, uint8_t *buf, unsigned max)
{
    unsigned n = 0, v;
    while (n < max && s[0] && s[1] && sscanf(s, "%2x", &v) == 1) {
        buf[n++] = (uint8_t)v;
        s += 2;
    }
    return n;
}

/* Parse "CC/N=HEX ... fill=FF sfdp=HEX" into s (s->log is left alone). */
static void sim_parse(struct sim *s, const char *spec)
{
    static char buf[16384];
    char *tok;
    FILE *log = s->log;
    memset(s, 0, sizeof *s);
    s->log = log;
    s->fill = 0xFF;
    strncpy(buf, spec, sizeof buf - 1);
    buf[sizeof buf - 1] = 0;
    for (tok = strtok(buf, " "); tok; tok = strtok(NULL, " ")) {
        unsigned cmd, txlen;
        char hex[64];
        if (!strncmp(tok, "fill=", 5)) {
            s->fill = (uint8_t)strtoul(tok + 5, NULL, 16);
        } else if (!strncmp(tok, "sfdp=", 5)) {
            s->sfdp_len = sim_hex(tok + 5, s->sfdp, sizeof s->sfdp);
        } else if (s->nreplies < 8 && sscanf(tok, "%2x/%u=%63s", &cmd, &txlen, hex) == 3) {
            struct sim_reply *r = &s->replies[s->nreplies++];
            r->cmd = (uint8_t)cmd;
            r->txlen = (uint8_t)txlen;
            r->len = (uint8_t)sim_hex(hex, r->data, sizeof r->data);
        }
    }
}

static void sim_log(FILE *f, const uint8_t *b, uint8_t n)
{
    uint8_t i;
    for (i = 0; i < n; i++)
        fprintf(f, "%02x", b[i]);
}

static void sim_xfer(void *ctx, const uint8_t *tx, uint8_t txlen, uint8_t *rx, uint8_t rxlen)
{
    struct sim *s = (struct sim *)ctx;
    unsigned i, k;
    for (i = 0; i < rxlen; i++)
        rx[i] = s->fill;
    if (tx[0] == 0x5A && txlen == 5 && s->sfdp_len) {
        /* RDSFDP: 3 address bytes, 8 dummy clocks (the fifth byte). */
        unsigned addr = (unsigned)tx[1] << 16 | (unsigned)tx[2] << 8 | tx[3];
        for (i = 0; i < rxlen; i++)
            rx[i] = addr + i < s->sfdp_len ? s->sfdp[addr + i] : 0xFF;
    } else {
        for (k = 0; k < s->nreplies; k++)
            if (s->replies[k].cmd == tx[0] && s->replies[k].txlen == txlen) {
                for (i = 0; i < rxlen && i < s->replies[k].len; i++)
                    rx[i] = s->replies[k].data[i];
                break;
            }
    }
    if (s->log) {
        fputs("> ", s->log);
        sim_log(s->log, tx, txlen);
        fputs(" < ", s->log);
        sim_log(s->log, rx, rxlen);
        fputc('\n', s->log);
    }
}

#if USF_HAVE_PROBE
/* Whether n bytes are all 0x00 or all 0xFF: nothing answered. */
static uint8_t usf__blank(const uint8_t *b, uint8_t n)
{
    uint8_t i, any = 0, all = 0xFF;
    for (i = 0; i < n; i++) {
        any |= b[i];
        all &= b[i];
    }
    return any == 0 || all == 0xFF;
}

/* Send `tx`, keep the `rxlen` bytes that come back in r, and look them up
 * as `family` unless the bus stayed blank. Returns r->count. */
static uint8_t usf__ask(usf_bus *bus, usf_probe_result *r, uint8_t family,
                        const uint8_t *tx, uint8_t txlen, uint8_t rxlen)
{
    USF_XFER(bus, tx, txlen, r->id, rxlen);
    r->family = family;
    r->len = rxlen;
    r->count = usf__blank(r->id, rxlen) ? 0 : usf_lookup(family, r->id, rxlen, r->chip);
    return r->count;
}

uint8_t usf_probe(usf_bus *bus, usf_probe_result *r)
{
    uint8_t tx[4], res[2], rdid[USF_RDID_LEN], i, saved = 0;

    tx[0] = 0xAB;                       /* RES: also wakes from deep power-down */
    tx[1] = tx[2] = tx[3] = 0;
    USF_XFER(bus, tx, 4, res, 2);

    tx[0] = 0x9F;                       /* JEDEC read-id */
    if (usf__ask(bus, r, USF_FAMILY_JEDEC, tx, 1, USF_RDID_LEN))
        return r->count;
    if (!usf__blank(r->id, USF_RDID_LEN)) {
        saved = 1;
        for (i = 0; i < USF_RDID_LEN; i++)
            rdid[i] = r->id[i];
    }
    /* SPI NAND: read-id after one dummy (or address) byte. */
    if (usf__ask(bus, r, USF_FAMILY_JEDEC, tx, 2, USF_RDID_LEN - 1))
        return r->count;

    tx[0] = 0x90;                       /* REMS: 3 address bytes */
    if (usf__ask(bus, r, USF_FAMILY_REMS, tx, 4, 2))
        return r->count;

    if (!usf__blank(res, 2)) {          /* the RES answer: two bytes, then one */
        r->id[0] = res[0];
        r->id[1] = res[1];
        r->family = USF_FAMILY_RES2;
        r->len = 2;
        if ((r->count = usf_lookup(USF_FAMILY_RES2, res, 2, r->chip)) != 0)
            return r->count;
        r->family = USF_FAMILY_RES1;
        r->len = 1;
        if ((r->count = usf_lookup(USF_FAMILY_RES1, res, 1, r->chip)) != 0)
            return r->count;
    }

    tx[0] = 0x15;                       /* Atmel AT25F read-id */
    if (usf__ask(bus, r, USF_FAMILY_AT25F, tx, 1, 2))
        return r->count;
    tx[0] = 0x83;                       /* ST M95 read-id: 2 address bytes */
    if (usf__ask(bus, r, USF_FAMILY_ST95, tx, 3, 3))
        return r->count;

    /* Nothing known answered: report the JEDEC bytes, if there were any. */
    r->family = USF_FAMILY_JEDEC;
    r->count = 0;
    r->len = 0;
    if (saved) {
        r->len = USF_RDID_LEN;
        for (i = 0; i < USF_RDID_LEN; i++)
            r->id[i] = rdid[i];
    }
    return 0;
}
#endif
